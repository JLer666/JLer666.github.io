进行统一的代码风格并进行代码静态检查，git commit时利用pre-commit检查,hook有clang-format,cppcheck,cpplint

sudo apt-get update
sudo pip install clang-format
sudo apt install python3-pkg-resources python3-pygments
sudo apt install cppcheck cppcheck-gui python3-setuptools ttf-bitstream-vera
sudo pip install cpplint
sudo pip install pre-commit

配置

cd到对应仓库 pre-commpreit install

cd到对应仓库
echo "repos:
  - repo: https://gitee.com/rxzh/pre-commit-hooks
    rev: v1.0
    hooks:
      - id: clang-format
        args: ["-i","--style=file"]
      - id: cppcheck
      - id: cpplint
        args: ["--linelength=120","--filter=-legal/copyright,-whitespace,-readability/fn_size, -build/include_what_you_use"]
" > .pre-commit-config.yaml

git add & git commit clang-format会自动修改不符合规定的代码，但是cppcheck及cpplint回检查出的问题需要手动修改，然后再次git add

内存安全检查，使用gcc自带的ASAN来做内存安全检查不用安装其他工具，只需要修改编译参数

add_compile_options(选择gcc)

add_compile_options(-fsanitize=address)
add_compile_options(-fno-omit-frame-pointer)
add_compile_options(-fsanitize-recover=address)
set(CMAKE_SHARED_LINKER_FLAGS -Wl,-Bsymbolic)

export ASAN_OPTIONS环境变量来设置内存检测的选项

export ASAN_OPTIONS=new_deleta_type_mismatch=0:halt_on_error=0
常用选项
halt_on_error=0													// 检测内存错误后继续运行默认是1 退出
delete_leaks=1													// 使能内存泄漏检测，默认不检测
new_delete_type_mismatch=0							// 忽略new delete类型不匹配
detect_stack_use_after_return=1					// 检查访问指向已被释放的栈空间，默认不检测
log_path=./asan.log											// 日志输出路径

分支上存在多次提交，使用git rebase -i [start commit] [end commit]进行合并，end commit 缺省时默认是HEAD.注意实际对应commit区间是(start commit,end commit]

pick # 保留提交不变
rewoed # 保留提交，但修改提交说明
edit # 保留提交,进入shell以便进行提交修补
squash # 保留提交，但融合到前一个提交
fixup # 类似于squash,但丢弃提交说明日志
exec # 使用shell运行命令
break # 在此处停止
drop # 删除提交
