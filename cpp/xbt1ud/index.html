2虚拟地址空间

| 3虚拟地址空间 |

| — |

虚拟地址空间实际不存在,是为了让地址不连续的空间能被利用起来。每个可执行程序被运行后(进程！！！为分配资源的最小单位)都对应一个虚拟内存空间，就是上面这幅图。空间大小由OS位数决定32位机大小为2^32 约为4G,64位机的大小为2^48。以32机为例，0-3G为用户空间，3G-4G(高地址)为内核空间。**每个虚拟地址空间的数据地址都会被CPU的MMU(内存管理单元)映射到真实的物理内存地址。**最多4G实际不会占用4G。NULL与nullptr的地址都在受保护地址。命令行参数为argc,argv,环境变量在linux中输入env就可看到所有环境变量。

普通用户对内核的读写权限都没有。

4文件描述符

| 4文件描述符 |

| — |

虚拟地址空间中的内核空间中，由这个进程(跑起来的程序)的信息PCB。PCB中有一个数组称为**文件描述符表(默认长1024)，**每个文件描述符都能定位1个文件，前三个符被默认占用标准输入输出错误默认状态为打开，这三个默认描述符对应三个文件，Linux中所有的东西都为文件，磁盘是文件，屏幕是文件，键盘是文件，显卡是文件，**所有的硬件设备最终都会被虚拟为一个文件(设备文件)，OS通过这个文件对硬件进行管理，用户可通过文件对程序进行管理。**这三个默认描述符对应的是当前终端/dev/tty设备文件。**同一个文件可被多次打开，多次打开后返回的FILE中的文件描述符的值是不一样的。**当使用fclose(c)或close(系统调用)时会将这个文件描述符释放等待下个文件打开再使用。文件描述符的分配规则是在表中找一个没被占用的最小值文件描述符给当前打开的文件。

/*
man 2 read
#include <unistd.h>//在这个头文件里 unix_stander
ssize_t read(int fd, void *buf, size_t count);
    参数:
     -fd:文件描述符 open的返回值 可通过文件描述符操作具体的一个函数（从命令行中读可设置为0）
     -buf:读到数据具体存放的地方(指定存在内存中这个地方) 给个数组首地址 数组具体多大自己给但小于要读的字节数会返回0
     -count:读取的字节数（是请求读取的字节数，读上来的数据保 存在buf中，同时文件的当前读写位置向后移）
    返回值:
     读成功会返回读到的数据的字节数，如果在调read之前已到达文件末尾，则这次read返回0,出现错误返回-1并设置errorno

     buf是个传出参数。指定读count个字节的数据，这个conut个字节的数据会被保存在buf中我们read后可再继续操作读出来的数据
     我们可以指定一个较小的buf(不需要buf必须等于或大于被读文件的大小)，指定读取sizeof(buf)个字节每次将buf给读满,然后我们将读到的数据拿去处理,处理完后再继续读，直到将文件读完

如果sizeof(buf)<count会导致文件被当写满buf后会继续往后写入野内存直到写满读到的总字节数这是很危险的，所以记住buf的大小要大于等于count，最后一次读到的数据可能会小于count因为读不满提前结束了返回0

     读常规文件是不会阻塞的，不管读多少字节，read一定会在有限的时间内返回。
     从终端设备或网络读则不一定，如果从终端输入的数据没有换行符，调用read读终端设备就会阻塞，如果网络上没有接收到数据包，调用read从网络读就会阻塞，至于会阻塞多长时间也是不确定的，
     如果一直没有数据到达就一直阻塞在那里。同样，写常规文件是不会阻塞的，而向终端设备或网络写则不一定。
     线程变为阻塞态
man 2 write
#include <unistd.h>
ssize_t write(int fd, const void *buf, size_t count);
    参数:
     -fd:文件描述符 open的返回值 可通过文件描述符操作具体的一个函数（输出(写)到命令行中可设置为1）
     -buf:常量指针 buf指向的数据不能修改(其实也符合write的逻辑 buf是要写入文件的内容
      相当于范本 范本怎么能改呢) 范本数据(往文件中具体要写的数据)
     -count:要写多少个字节的数据 从buf中取多少个字节数据向fd文件中写
    返回值:
      成功返回buf中被写入fd文件的字节数，返回0表示没有数据被写入fd文件中,出现错误返回-1并设置errorno
      如果count大于sizeof(buf)，那么会把缓冲区后面的内存中的数据写进去，只不过这些数据我们是不确定的，是野内存，
      操作野内存有可能会产生问题，所以一般不会这么去做。缓冲区中有多少数据，我们就写多少数据即可。

#include <stdio.h>
#include <sys/types.h> //open flags宏在这个头文件

int main()
{
    //通过open打开english.txt
    int srcfd = open("./english.txt", O_RDONLY); //打开源文件 只给读的权限
    if (srcfd == -1)
    {
        perror("open_fail:");
        return -1;
    }
    //创建一个新的文件
    int dstfd = open("./dst.txt", O_WRONLY | O_CREAT, 0664); //只写和创造
    if (dstfd == -1)
    {
        perror("open_fail:");
        return -1;
    }
    //读写
    char buf[2] = {0}; //故意给个很小的buf 指定一个较小的buf(不需要buf必须等于或大于被读文件的大小)
    int read_how_many_bits = 0;

	while ((read_how_many_bits = read(srcfd, buf, sizeof(buf))) > 0) //read返回-1表示错误 返回0表示读完
    {                                                                //大于0则表示文件还没读完

        //read_how_many_bits就是实际读到多少字节 读到多少就写多少
    }
    //关闭文件
    close(dstfd);
    close(srcfd);

    //gcc txt_copy.c -o copy

对应标准c –seek

/*
  标准c fseek
  #include <stdio.h>
  int fseek(FILE *stream, long offset, int whence);

  linux系统调用
  #include <sys/types.h>//宏
  #include <unistd.h>//unix标准中声明
  off_t lseek(int fd, off_t offset, int whence);
  通过lseek对文件中的文件读写指针进行移动
   参数
    -fd open得到的文件描述符
    -offset 偏移量
    -whence 指定的标记 (和标准C中指定值是一样的)
       SEEK_SET 参数offset 即为新的读写位置.
              The file offset is set to offset bytes.
       SEEK_CUR 以目前的读写位置往后增加offset 个位移量.
              The file offset is set to its current location plus offset bytes.
       SEEK_END 将读写位置指向文件尾后再增加offset 个位移量
              The file offset is set to the size of the file plus offset bytes.
       当whence 值为SEEK_CUR 或SEEK_END 时, 参数offet 允许负值的出现.
   返回值
       返回偏移后的文件指针位置

    欲将读写位置移到文件开头时:lseek( fd, 0, SEEK_SET);
    欲将读写位置移到文件尾时:lseek( fd, 0, SEEK_END);//可以直接获得这个文件的大小
    想要取得目前文件位置时:lseek( fd, 0, SEEK_CUR);
    拓展文件的大小 lseek( fd, 100, SEEK_END);//为文件增加100字节的大小
    比如在下载时提前占用要下载文件大小的空间以免被其他进程占用，之后再慢慢往文件中写数据
\*/
//通过ll查看english.txt文件的大小为54个字节
#include <sys/types.h> //open flags宏在这个头文件 lseek宏在这个头文件
#include <sys/stat.h>  //open flags宏在这个头文件里也有
#include <fcntl.h>     //open  close函数声明在这个头文件中
#include <unistd.h>    //unix标准中声明
#include <stdio.h>
//gcc lseek.c -o lseek
int main()
{
    int fd = open("english.txt", O_RDWR);
    if (fd == -1)
    {
        perror("open:");
        return -1;
    }
    //拓展文件的大小
    int ret = lseek(fd, 100, SEEK_END);
    if (ret == -1)
    {
        perror("lseek:");
        return -1;
    }
    //写入空数据 需要写入一次数据 拓展的大小才会生效要不然还是原来那么大
    write(fd, "", 1);
    //155 = 54 + 100 +写入的空(1)
    close(fd);
    return 0;
}

               dev_t     st_dev;//文件的设备编号
               ino_t     st_ino;//节点 Inode
               mode_t    st_mode;//文件的类型和存取权限
               //st_mode为16位整数 具体看图
               nlink_t   st_nlink;//连接到该文件的硬连接数
               uid_t     st_uid;//用户ID
               gid_t     st_gid;//组ID
               dev_t     st_rdev;//设备文件的设备编号
               off_t     st_size;//文件大小 字节数
               blksize_t st_blksize;//块大小 每块512字节
               blkcnt_t  st_blocks; //块数
            struct timespec st_atim;//最后一次访问时间
            struct timespec st_mtim;//最后一次修改时间
            struct timespec st_ctim;//最后一次改变时间 指属性
            #define st_atime st_atim.tv_sec
            #define st_mtime st_mtim.tv_sec
            #define st_ctime st_ctim.tv_sec
           };

#include <sys/types.h>
#include <sys/stat.h>
#include <unistd.h>
#include <stdio.h>
#include <pwd.h>    //getpwuid
#include <grp.h>    //getgrgid
#include <time.h>   //ctime
#include <string.h> //strncpy strlen
int main(int argc, char *argv[])
{
    if (argc < 2)
    {
        //一般一定会有一个参数 是可执行文件的路径
        //再加上跟上的文件名 一共至少为两个参数 模拟才会生效
        //所以这里判断一下参数个数小于2个就return
        printf("%s filename\n", argv[0]); //argv[0]为执行文件的路径
        return -1;
    }
    struct stat statbuf;
    //通过stat获得文件的参数
    int ret = stat(argv[1], &statbuf); //argv[1]为传入的文件路径
    if (ret == -1)
    {
        perror("stat:");
        return -1;
    }
    //获取文件类型和文件权限
    char perms[11] = {0}; //1位文件类型+9位文件权限+一位/0
    //具体文件类型和权限在st_mode中可以具体看word 22页的图
    // (st_mode & S_IFMT) == 要判断的文件类型宏
    // S_IFMT = 0170000(8进制)-- > 1111 000 000 000 000 只筛选出前四位来判断(前四位就指定了文件类型)
    //
    switch (statbuf.st_mode & S_IFMT) //
    {
    case S_IFLNK: //链接文件
        perms[0] = 'l';
        break;
    case S_IFDIR: //目录
        perms[0] = 'd';
        break;
    case S_IFREG: //普通文件
        perms[0] = '-';
        break;
    case S_IFBLK: //块文件
        perms[0] = 'b';
        break;
    case S_IFCHR: //字符设备
        perms[0] = 'c';
        break;
    case S_IFSOCK: //套接字文件
        perms[0] = 's';
        break;
    case S_IFIFO: //管道文件
        perms[0] = 'p';
        break;
    default:
        perms[0] = '?'; //未知文件
        break;
    }
    //获取文件的访问权限分为3组
    //文件所有者
    perms[1] = statbuf.st_mode & S_IRUSR ? 'r' : '-'; //用户读权限 有这个权限的话按位与表达式不为0
    perms[2] = statbuf.st_mode & S_IWUSR ? 'w' : '-'; //用户写权限 有这个权限的话按位与表达式不为0
    perms[3] = statbuf.st_mode & S_IXUSR ? 'x' : '-'; //用户执行权限 有这个权限的话按位与表达式不为0
    //文件所在组权限
    perms[4] = statbuf.st_mode & S_IRGRP ? 'r' : '-'; //所在组读权限 有这个权限的话按位与表达式不为0
    perms[5] = statbuf.st_mode & S_IWGRP ? 'w' : '-'; //所在组写权限 有这个权限的话按位与表达式不为0
    perms[6] = statbuf.st_mode & S_IXGRP ? 'x' : '-'; //所在组执行权限 有这个权限的话按位与表达式不为0
    //其他人
    perms[7] = statbuf.st_mode & S_IROTH ? 'r' : '-'; //其他人读权限 有这个权限的话按位与表达式不为0
    perms[8] = statbuf.st_mode & S_IWOTH ? 'w' : '-'; //其他人写权限 有这个权限的话按位与表达式不为0
    perms[9] = statbuf.st_mode & S_IXOTH ? 'x' : '-'; //其他人执行权限 有这个权限的话按位与表达式不为0
    //硬连接数
    int linknum = statbuf.st_nlink;
    //文件所有者
    // extern struct passwd *getpwuid(__uid_t __uid);
    // struct passwd
    // {
    //     char *pw_name;   /* Username. 我们需要的用户名称  */
    //     char *pw_passwd; /* Password.  */
    //     __uid_t pw_uid;  /* User ID.  */
    //     __gid_t pw_gid;  /* Group ID.  */
    //     char *pw_gecos;  /* Real name.  */
    //     char *pw_dir;    /* Home directory.  */
    //     char *pw_shell;  /* Shell program.  */
    // };
    char *fileUser = getpwuid(statbuf.st_uid)->pw_name; //只能得到uid而非用户名称 需要通过getpwuid得到
    //文件所在组 同理
    char *fileGrp = getgrgid(statbuf.st_gid)->gr_name;
    //文件大小
    long int fileSize = statbuf.st_size;
    //获取修改时间
    char *time = ctime(&statbuf.st_mtime); //从1970 1.1 0:0到当前的秒数 通过ctime变为当前时间
    //时间自带换行 要将这个换行去掉
    char mtime[512] = {0};
    strncpy(mtime, time, strlen(time) - 1); //-1就是去掉最后的这个换行符
    char buf[1024];
    sprintf(buf, "%s %d %s %s %ld %s %s", perms, linknum, fileUser, fileGrp, fileSize, mtime, argv[1]);
    printf("%s\n", buf);
    //向buf中添加内容 %s文件类型和文件权限 %d连接数 %s所有者 %s所在组 %d文件大小 %s修改时间 %s文件名
    return 0;
}
