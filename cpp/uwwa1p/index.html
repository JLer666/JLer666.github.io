3 GUN2.9 std::alloc源码分析

//上调传入的bytes到它最接近的8的整数倍
static size_t ROUND_UP(size_t bytes)
{
    return (((bytes) + __ALIGN - 1) & ~(__ALIGN - 1));
    //举例 如果bytes = 13
    //则 (13+7)&~7 --- 10100 & ~(00111)
    //= 10100 & 11000 = 10000 = 16  ( 其实bytes是8位的 这里只写出5位)
}

图中的灰色部分就是obj的free_list_link指针所占的大小，灰色+白色的总的空间是我们申请(分割出的)的一个对象的空间(sizeof(对象))，一次性申请了n*sizeof(对象的大小)。(free_list_link 就是next)

//p = (obj *)malloc(n*sizeof(对象));
for (int i = 0; i < n - 1; i++)
{
    p->free_list_link= (obj *)((char *)p + sizeof(对象)); //将空间强转为obj型
    //size为被管理类 的一个对象的大小！！！！ 单位字节 所以要将p转为char型 也变为以字节为单位
    p = p->free_list_link;
}
p->free_list_link= nullptr; //last

不断地通过p+sizeof(对象)定位到精确的每个对象的起始地址。并将这个对象强转为obj类型，这样的话这块首地址为p+sizeof(对象)大小为sizeof(对象)空间的前四个字节就变为了free_list_link指针(将对象变量–sizeof(对象)这种类型大小比obj大的变量强制转换为 obj类型– 类似将double型变量强转为float型变量的感觉，这种将较大数据类型的变量强转为较小数据类型的变量 之前用的很少。如果我们的对象的大小小于4字节，是保存不了obj类型的free_list_link指针4字节将会出错–就是上面那张图白色部分很小，灰色部分超出了白色部分,发生内存越界)，可以用于连接其他对象节点的首地址，这样就实现了精确分割出了一个对象的内存并且可以通过free_list_link指针指针将它们连成一个链表用于管理。

// chunk = szieof(对象) * n
// p = (obj *)malloc(chunk);
// p指针是在不断移动的 不断地指向每一个size大小内存块的首地址
//(p其实指向的是size大小空间的前四个字节组成的obj的首地址)  next指针就是free_list_link
// |next指针 .... 这块有size大|...next指针 .... 这块有size大|....|
// 每块空间首地址被强转为obj型 所以空间中的前4字节暂时被当作next指针 串联其他size大小的空间
//      |-------------------------|(next指向下一块的首地址)
// 只有在分配和回收的时候我们才需要这个next指针
// 在allocate return了这块空间的首地址后 内存块已经不在链表里了next指针没有意义了
// 这四个字节可以去存真正的对象内容
// 将每一块串成list
union obj
{
    union obj *free_list_link;
};

obj* freelist就是链表数组用于管理那16个链表，FREELIST_INDEX函数 传入的比特值返回管理这个比特值的链表在链表数组中的索引。(8->0 16->1 20->2(向上找 20字节的节点不能由16字节的链表管理，是由24字节的链表管理) 24->2)。refill函数实现的是之前讲的free_list[i]管理的区块不够了先看战备池空间是否够，从战备池中取取出最多20个对象大小的空间(少于20则能分几个就分几个 1个都分不出来则就是碎片)，战备池是否是碎片，将碎片交给对应的链表管理，之后malloc一块大内存，并串到free_list[i]这个链表的这个过程。

chunk_alloc是refill中要用到的函数，申请n块大小为size的内存，start_free,end_free用于指向我们之前说过的战备池的头和尾，heap_size是前面说的累计分配量(用于计算追加分配量)。

 // n为一个对象的大小(字节)
static void *allocate(size_t n) //n must be > 0
{
    obj *volatile *my_free_list; //obj** my_free_list; 用于指向16条链表中一条的首地址
    obj *result;

    //申请的大小大于128字节 调用第一级分配器 直接用malloc来分配
    if (n > (size_t)__MAX_BYTES)
    {
        return (malloc_alloc::allocate(n));
    }
    my_free_list = free_list + FREELIST_INDEX(n);
    //my_free_list指向 指向第n条链表头节点的指针(my_free_list为二级指针)
    result = *my_free_list;
    if (result == 0) //一个指向第n条链表头节点的指针
    {
        //这个指针为0说明 此指针并没有指向链表节点---这条链表内内存节点已经用完了
        //需要再次分配 去malloc n*20*2+ROUNDUP(HEAPSIZE>>4)的空间给这条链表
        void *r = refill(ROUND_UP(n));
        return r;
    }
    //result此指针非空 说明result指向的这条链表至少还有一个内存节点(链表头节点)
    //则返回这个链表头节点给用户
    //头节点指针指向头节点的下个节点作为新的头节点
    *my_free_list = result->free_list_link;
    return (result);
}


//回收对象内存块 p为这个对象内存的首地址 n为这个对象的大小
//将这个对象节点 回收到管理对应大小的链表的头节点处(作为链表新的头节点的方式 回收回链表中)
static void deallocate(void *p, size_t n) //p may not be 0
{
    obj *q = (obj *)p;
    obj *volatile *my_free_list; //obj** my_free_list;

    //大小大于128字节 不由第二级分配器管理 直接调用第一级deallocate 由free来回收
    if (n > (size_t)__MAX_BYTES)
    {
        malloc_alloc::deallocate(p, n);
        return;
    }
    //找到管理对应大小的 这条链表的头节点的地址*my_free_list
    //将回收回来的这个节点p的next指针指向这个头节点
    //并将头节点指针(*my_free_list)指向p  这样p就是这条链表的新的头节点
    my_free_list = free_list + FREELIST_INDEX(n);
    q->free_list_link = *my_free_list;
    *my_free_list = q;
}

这种deallocate行为将会造成下面2个问题

** 1注意在第二级deallocate中并没有调用free 这意味着16条链表将会无限拉，deallocate不会将内存还给操作系统 第二级分配器不会将内存还给操作系统，但是电脑上并不会只跑你这一个程序 内存都在你手上 那其他程序怎么办呢? 因为没有记录每次malloc得到的带cookie的上下两块内存的地址 所以没办法通过delete or free来归还**

2如果我不是用第二级分配器申请的内存，比如用直接malloc申请了16个字节，这16个字节的内存是带cookie的。但是我调用了第二级分配器的deallocate 是可以将这个malloc申请的内存归还到对应大小的链表中 (带cookie并不会有太大的影响 指针是不指向cookie的)，但是如果归还的这块内存 不是用第二级分配器申请的非8的整数倍的内存归还时回被还到 其最接近且大于它的8的整数倍的链表中，在某次第二级allocate会将这块实际没那么大的内存 当作有这么大交给用户去使用 将会造成内存泄漏。

refill函数行为

申请20个区块(不一定有20个的空间 具体申请到多少个由chunk_alloc决定—-调用chunk申请20个，chunk还会多申请20个的空间给战备池)，如果chunk_alloc只拿到1个对象的空间则直接把这个内存空间返回给用户，如果有多余1个，则将余下的对象空间节点挂到对应大小的free_list链表上。

template <bool threads, int inst>
void *__default_alloc_template<threads, inst>::
refill(size_t n) //n为字节大小 已经上调为8的倍数
{
    int nobjs = 20; //申请20个对象大小的内存 1个返回用户 19给链表 实际是40个多余的20个加入战备池中
    //预设要20个 但不一定能取到20个 nobjs是引用在申请完后会返回究竟申请多少个对象大小的内存
    char *chunk = chunk_alloc(n, nobjs); //传回nobjs个对象大小的内存(期望是20个)
    obj *volatile *my_free_list;         //obj** my_free_list;
    obj *result;
    obj *current_obj;
    obj *next_obj;
    int i;

    if (1 == nobjs) //实际只拿到1个对象大小的内存 则将这块内存直接返回给用户
        return (chunk);
    //运行到这里说明有不止1块 1块给用户后还有多余 则多余的交给对应大小的链表管理

    //找到管理对应大小的 这条链表的头节点的地址*my_free_list
    my_free_list = free_list + FREELIST_INDEX(n);
    //Build free list in chunk
    //将chunk内多余的内存 挂到对应的链表上
    result = (obj *)chunk; //大块 中的第一个(对象大小)的区块 作为result返回给用户
    //余下的区块要串到链表上
    *my_free_list = next_obj = (obj *)(chunk + n); //第二个区块的首地址
    for (i = 1;; ++i)                              //从1开始将返回给用户的那个排除
    {
        current_obj = next_obj;                   //上次的下一块 就是当前块
        next_obj = (obj *)((char *)next_obj + n); //下一块
        //因为拿到的内存是一整块连续的 干净的内存没什么对象变量
        //我们需要切割成一块又一块 大小为n的区块并将它们串起来
        //所谓切割就是将指针所指处+n个字节先使其指向下个区块的首地址
        //再转型为obj(使大小为n区块的前四字节变为指针 可指向下一个区块
        //从而实现串联)
        if (nobjs - 1 == i)
        {
            //最后一块 其next指针指向null 之后结束循环
            current_obj->free_list_link = 0;
            break;
        }
        else
        {
            current_obj->free_list_link = next_obj; //当前块指向下一块 串成单向链表
        }
    }
    return (result);
}
