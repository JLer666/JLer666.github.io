    //放在 namespace 中因此和 std 不衝突

#define __THROW_BAD_ALLOC            \
    cerr << "out of memory" << endl; \
    exit(1)
    //----------------------------------------------
    // 第1級配置器。
    //----------------------------------------------
    template <int inst>
    //申请的内存大于128字节 调用第一级分配器 第一级分配器直接调用 malloc从堆中拿内存
    //没有其他多余出来
    //当第二级分配器 申请内存失败后 会调用第一级分配器的oom_realloc循环申请内存 直到拿到内存
    //
    class __malloc_alloc_template
    {
    private:
        static void *oom_malloc(size_t);
        static void *oom_realloc(void *, size_t);
        //老版本c++ 新版本调用的是::operator new() 对应的无内存处理是set_new_handler()
        //在新的gnu版本中不存在第一级分配器
        static void (*__malloc_alloc_oom_handler)();

    public:
        static void *allocate(size_t n)
        {
            void *result = malloc(n); //直接使用 malloc()
            if (0 == result)
                result = oom_malloc(n);
            return result;
        }
        static void deallocate(void *p, size_t /* n */)
        {
            free(p); //直接使用 free()
        }
        static void *reallocate(void *p, size_t /* old_sz */, size_t new_sz)
        {
            void *result = realloc(p, new_sz); //直接使用 realloc()
            if (0 == result)
                result = oom_realloc(p, new_sz);
            return result;
        }
        static void (*set_malloc_handler(void (*f)()))()
        { //類似 C++ 的 set_new_handler().
            void (*old)() = __malloc_alloc_oom_handler;
            __malloc_alloc_oom_handler = f;
            return (old);
        }
    };
    //----------------------------------------------
    template <int inst>
    void (*__malloc_alloc_template<inst>::__malloc_alloc_oom_handler)() = 0;

    template <int inst>
    void *__malloc_alloc_template<inst>::oom_malloc(size_t n)
    {
        void (*my_malloc_handler)();
        void *result;

        for (;;)
        { //不斷嘗試釋放、配置、再釋放、再配置…
            my_malloc_handler = __malloc_alloc_oom_handler;
            if (0 == my_malloc_handler)
            {
                __THROW_BAD_ALLOC;
            }
            (*my_malloc_handler)(); //呼叫處理常式，企圖釋放記憶體
            result = malloc(n);     //再次嘗試配置記憶體
            if (result)
                return (result);
        }
    }

    template <int inst>
    void *__malloc_alloc_template<inst>::oom_realloc(void *p, size_t n)
    {
        void (*my_malloc_handler)();
        void *result;

        for (;;)
        { //不斷嘗試釋放、配置、再釋放、再配置…
            my_malloc_handler = __malloc_alloc_oom_handler;
            if (0 == my_malloc_handler)
            {
                __THROW_BAD_ALLOC;
            }
            (*my_malloc_handler)(); //呼叫處理常式，企圖釋放記憶體。
            result = realloc(p, n); //再次嘗試配置記憶體。
            if (result)
                return (result);
        }
    }

    template <class T, class Alloc>
    class simple_alloc
    {
    public:
        static T *allocate(size_t n)
        {
            return 0 == n ? 0 : (T *)Alloc::allocate(n * sizeof(T));
        }
        static T *allocate(void)
        {
            return (T *)Alloc::allocate(sizeof(T));
        }
        static void deallocate(T *p, size_t n)
        {
            if (0 != n)
                Alloc::deallocate(p, n * sizeof(T));
        }
        static void deallocate(T *p)
        {
            Alloc::deallocate(p, sizeof(T));
        }
    };

    enum
    {
        __ALIGN = 8
    }; //小區塊的上調邊界 ROUNDUP时向上8字节对齐 链表按8字节为间隔区分管理
    enum
    {
        __MAX_BYTES = 128
    }; //小區塊的上限
    //8-128字节的申请才会使用第二级分配器
    enum
    {
        __NFREELISTS = __MAX_BYTES / __ALIGN
    }; //free-lists 個數 16条链表

    //上面的这三个枚举量因该用 static const int x = N
    //取代 enum { x = N }, 但目前支援該性質的編譯器不多

    //本例中兩個 template 參數完全沒有派上用場
    template <bool threads, int inst>
    class __default_alloc_template
    {
    private:
        //上调传入的bytes到它最接近的8的整数倍
        static size_t ROUND_UP(size_t bytes)
        {
            return (((bytes) + __ALIGN - 1) & ~(__ALIGN - 1));
            //举例 如果bytes = 13
            //则 (13+7)&~7 --- 10100 & ~(00111)
            //= 10100 & 11000 = 10000 = 16  ( 其实bytes是8位的 这里只写出5位)
        }

    private:
        //改用struct 也可以
        //嵌入式指针
        //用struct更好理解
        //就像链表中的next指针一样 每个obj对象只有1个成员变量就是指针 可以指向obj类的其他对象

        //注意 我们只知道 我们要分配的单个对象的大小 但是不知道我们要分配对象的类型
        //所以这里malloc没办法强转为那个对象的类型  但是知道大小一样也可以将这一大块空间
        //分割为多个 szieof(对象)的小块并用指针(上个小块的next指针)指向这小块的首地址 串成一个链表

        // chunk = szieof(对象) * n
        // p = (obj *)malloc(chunk);
        // p指针是在不断移动的 不断地指向每一个size大小内存块的首地址
        //(p其实指向的是size大小空间的前四个字节组成的obj的首地址)  next指针就是free_list_link
        // |next指针 .... 这块有size大|...next指针 .... 这块有size大|....|
        // 每块空间首地址被强转为obj型 所以空间中的前4字节暂时被当作next指针 串联其他size大小的空间
        //      |-------------------------|(next指向下一块的首地址)
        // 只有在分配和回收的时候我们才需要这个next指针
        // 在allocate return了这块空间的首地址后 内存块已经不在链表里了next指针没有意义了
        // 这四个字节可以去存真正的对象内容
        // 将每一块串成list
        union obj
        {
            union obj *free_list_link;
        };

    private:
        static obj *volatile free_list[__NFREELISTS];
        static size_t FREELIST_INDEX(size_t bytes)
        {
            return (((bytes) + __ALIGN - 1) / __ALIGN - 1);
        }

        // Returns an object of size n, and optionally adds to size n free list.
        static void *refill(size_t n);

        // Allocates a chunk for nobjs of size "size".  nobjs may be reduced
        // if it is inconvenient to allocate the requested number.
        static char *chunk_alloc(size_t size, int &nobjs);

        // Chunk allocation state.
        static char *start_free;
        static char *end_free;
        static size_t heap_size;

    public:
        // n为一个对象的大小(字节)
        static void *allocate(size_t n) //n must be > 0
        {
            obj *volatile *my_free_list; //obj** my_free_list; 用于指向16条链表中一条的首地址
            obj *result;

            //申请的大小大于128字节 调用第一级分配器 直接用malloc来分配
            if (n > (size_t)__MAX_BYTES)
            {
                return (malloc_alloc::allocate(n));
            }

            my_free_list = free_list + FREELIST_INDEX(n);
            //my_free_list指向 指向第n条链表头节点的指针(my_free_list为二级指针)
            result = *my_free_list;
            if (result == 0) //一个指向第n条链表头节点的指针
            {
                //这个指针为0说明 此指针并没有指向链表节点---这条链表内内存节点已经用完了
                //需要再次分配 去malloc n*20*2+ROUNDUP(HEAPSIZE>>4)的空间给这条链表
                void *r = refill(ROUND_UP(n));
                return r;
            }
            //result此指针非空 说明result指向的这条链表至少还有一个内存节点(链表头节点)
            //则返回这个链表头节点给用户
            //头节点指针指向头节点的下个节点作为新的头节点
            *my_free_list = result->free_list_link;
            return (result);
        }

        //回收对象内存块 p为这个对象内存的首地址 n为这个对象的大小
        //将这个对象节点 回收到管理对应大小的链表的头节点处(作为链表新的头节点的方式 回收回链表中)
        static void deallocate(void *p, size_t n) //p may not be 0
        {
            obj *q = (obj *)p;
            obj *volatile *my_free_list; //obj** my_free_list;

            //大小大于128字节 不由第二级分配器管理 直接调用第一级deallocate 由free来回收
            if (n > (size_t)__MAX_BYTES)
            {
                malloc_alloc::deallocate(p, n);
                return;
            }
            //找到管理对应大小的 这条链表的头节点的地址*my_free_list
            //将回收回来的这个节点p的next指针指向这个头节点
            //并将头节点指针(*my_free_list)指向p  这样p就是这条链表的新的头节点
            my_free_list = free_list + FREELIST_INDEX(n);
            q->free_list_link = *my_free_list;
            *my_free_list = q;
        }
        //1
        //注意在第二级deallocate中并没有调用free 这意味着16条链表将会无限拉长
        //deallocate不会将内存还给操作系统 第二级分配器不会将内存还给操作系统
        //但是电脑上并不会只跑你这一个程序 内存都在你手上 那其他程序怎么办呢?

        //2
        //如果我不是用第二级分配器申请的内存，比如用直接malloc申请了16个字节
        //这16个字节的内存是带cookie的
        //但是我调用了第二级分配器的deallocate 是可以将这个malloc申请的内存
        //归还到对应大小的链表中 (带cookie并不会有太大的影响 指针是不指向cookie的)
        //但是如果归还的这块内存 不是用第二级分配器申请的非8的整数倍的内存
        //归还时回被还到 其最接近且大于它的8的整数倍的链表中
        //在某次第二级allocate会将这块实际没那么大的内存 当作有这么大交给用户去使用 将会造成内存泄漏

        static void *
        reallocate(void *p, size_t old_sz, size_t new_sz);
    };
    //----------------------------------------------
    // We allocate memory in large chunks in order to
    // avoid fragmentingthe malloc heap too much.
    // We assume that size is properly aligned.
    // We hold the allocation lock.
    //----------------------------------------------
    template <bool threads, int inst>
    char *
    __default_alloc_template<threads, inst>::
        chunk_alloc(size_t size, int &nobjs)
    {
        char *result;
        size_t total_bytes = size * nobjs;         //size为一个对象的大小 n为20
        size_t bytes_left = end_free - start_free; //当前战备池内剩下多少字节

        if (bytes_left >= total_bytes) //当前战备池内的大小满足 20个对象这个需求
        {
            //则返回这20个对象大小的内存
            //并减少对应的战备池大小
            result = start_free;
            start_free += total_bytes; //减少战备池大小
            //地址是向下增长的 start_free<=end_free
            return (result); //result 为原start_free-start_free+total_bytes这块连续空间
        }
        else if (bytes_left >= size) //当前战备池大小至少能分配一个对象的内存
        {
            // 先计算余下大小到底能分配多少个对象
            // 注意nobjs为引用传入在这里传出后 调用者就能知道究竟是分配了多少个
            // 对象的内存
            nobjs = bytes_left / size;
            total_bytes = size * nobjs;
            //余下的内存最多分配nobjs个对象 这nobjs个对象的总大小为total_bytes
            result = start_free;
            start_free += total_bytes; //减少战备池大小
            return (result);           //result 为原start_free-start_free+total_bytes这块连续空间
        }
        else
        {
            //当前战备池大小连一个对象大小的内存空间都不够了
            //则需要去向系统申请新的内存空间 并且将当前pool作为碎片交给对应的链表管理
            //申请--(我们之前一直有提到的 2*20*sizeof(对象) + ROUNDUP(累计申请量>>4))
            //其中20*sizeof(对象)的空间返回给调用者，余下的交给内存池
            size_t bytes_to_get =
                2 * total_bytes + ROUND_UP(heap_size >> 4);
            // Try to make use of the left-over piece.
            if (bytes_left > 0) //当前pool是有内存的并非刚好用完 则这些内存就是碎片要交给对应的链表管理
            {
                //将碎片交给对应的链表管理
                obj *volatile *my_free_list =
                    free_list + FREELIST_INDEX(bytes_left);
                //start_free就是碎片的起始地址
                ((obj *)start_free)->free_list_link = *my_free_list; //将碎片插到对应链表的头之前
                *my_free_list = (obj *)start_free;                   //将碎片作为新的头节点
            }
            start_free = (char *)malloc(bytes_to_get); //去 malloc 2*20*sizeof(对象) + ROUNDUP(累计申请量>>4)
            //将新申请的这块内存的首地址 作为pool的新的起始地址start_free

            if (0 == start_free) //malloc 申请失败会返回0
            {
                //应对堆中内存不够 malloc申请失败的情况
                int i;
                obj *volatile *my_free_list, *p;

                //Try to make do with what we have. That can't
                //hurt. We do not try smaller requests, since that tends
                //to result in disaster on multi-process machines.
                //找当前size对应链表的右边的链表(右边链表中的节点内存都是大于size的) 看右边链表中是否
                //还有内存节点 有的话应急直接拿来用 切出1个size大内存返回给用户 余下的作为pool战备池
                //例如szie=80的话 -- 向右边 查看管理88,96,104,112大小的节点
                //因该从size+__ALIGN开始循环更合理，因为size链表肯定没节点 要不然也不会到当前这个地步
                for (i = size; i <= __MAX_BYTES; i += __ALIGN)
                {
                    my_free_list = free_list + FREELIST_INDEX(i);
                    p = *my_free_list;
                    if (0 != p) //有右边的链表 有多余的节点 我们拿来借用
                    {
                        //p为我们借用的那块内存节点的首地址(也是那条链表头节点的首地址)
                        *my_free_list = p->free_list_link; //p被借用了 那条链表更新头节点为下个节点

                        //借来的这块空间 放到战备池中！！！！ start_free指向这块首节地址
                        start_free = (char *)p;
                        end_free = start_free + i;
                        return (chunk_alloc(size, nobjs));
                        //进行递归调用 在下一层的调用中 就可以直接满足我们的要求
                        //在上面的两个if就return了对应空间的大小
                        //这个递归只会向下一层就return 不会继续深入了
                        //Any leftover piece will eventually make it to the
                        //right free list.
                    }
                }
                //右边的链表中也没有内存节点(别人也破产了 借不到东西) 穷途末路了
                //拿不到内存 调用第一级分配器 第一级分配器会调用handler(out of memory)循环请求内存
                //看是否有进程/线程刚好释放内存则可马上拿来填填补这个需求
                end_free = 0; //In case of exception.
                start_free = (char *)malloc_alloc::allocate(bytes_to_get);
                //This should either throw an exception or
                //remedy the situation. Thus we assume it
                //succeeded.
            }
            // malloc分配成功 分配了我们要求的bytes_to_get这样大量的内存
            heap_size += bytes_to_get;            //累计申请量
            end_free = start_free + bytes_to_get; //战备池池底
            //成功申请了bytes_left
            //进行递归调用 在下一层的调用中 就可以直接满足我们的要求
            //在上面的两个if就return了对应空间的大小
            //这个递归只会向下一层就return 不会继续深入了
            return (chunk_alloc(size, nobjs));
        }
    }
    //----------------------------------------------
    // Returns an object of size n, and optionally adds
    // to size n free list.We assume that n is properly aligned.
    // We hold the allocation lock.
    //----------------------------------------------
    template <bool threads, int inst>
    void *__default_alloc_template<threads, inst>::
        refill(size_t n) //n为字节大小 已经上调为8的倍数
    {
        int nobjs = 20; //申请40个对象大小的内存 1个返回用户 19给链表 20个加入战备池中
        //预设要20个 但不一定能取到20个 nobjs是引用在申请完后会返回究竟申请多少个对象大小的内存
        char *chunk = chunk_alloc(n, nobjs); //传回nobjs个对象大小的内存(期望是20个)
        obj *volatile *my_free_list;         //obj** my_free_list;
        obj *result;
        obj *current_obj;
        obj *next_obj;
        int i;

        if (1 == nobjs) //实际只拿到1个对象大小的内存 则将这块内存直接返回给用户
            return (chunk);
        //运行到这里说明有不止1块 1块给用户后还有多余 则多余的交给对应大小的链表管理

        //找到管理对应大小的 这条链表的头节点的地址*my_free_list
        my_free_list = free_list + FREELIST_INDEX(n);

        //Build free list in chunk
        //将chunk内多余的内存 挂到对应的链表上
        result = (obj *)chunk; //大块 中的第一个(对象大小)的区块 作为result返回给用户
        //余下的区块要串到链表上
        *my_free_list = next_obj = (obj *)(chunk + n); //第二个区块的首地址
        for (i = 1;; ++i)                              //从1开始将返回给用户的那个排除
        {
            current_obj = next_obj;                   //上次的下一块 就是当前块
            next_obj = (obj *)((char *)next_obj + n); //下一块
            //因为拿到的内存是一整块连续的 干净的内存没什么对象变量
            //我们需要切割成一块又一块 大小为n的区块并将它们串起来
            //所谓切割就是将指针所指处+n个字节先使其指向下个区块的首地址
            //再转型为obj(使大小为n区块的前四字节变为指针 可指向下一个区块
            //从而实现串联)
            if (nobjs - 1 == i)
            {
                //最后一块 其next指针指向null 之后结束循环
                current_obj->free_list_link = 0;
                break;
            }
            else
            {
                current_obj->free_list_link = next_obj; //当前块指向下一块 串成单向链表
            }
        }
        return (result);
    }

    void test_G29_alloc()
    {
        cout << "\n\n\ntest_global_allocator_with_16_freelist().......... \n";

        void *p1 = alloc::allocate(120);
        void *p2 = alloc::allocate(72);
