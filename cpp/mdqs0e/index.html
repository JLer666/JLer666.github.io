当某个Chunk全回收了(分出去的区块都收回来了)，会进行判断当前是否有两个Chunk都是全回收状态(如果只有一个Chunk全回收，有很大的几率下次又会有使用者申请区块，所以如果只有一个Chunk全回收后，直接释放这个Chunk的内存 在下次又有申请请求到了，又需要再去malloc一个Chunk这回造成很大的不便,所以只在出现两个全回收Chunk的情况下我们将老的那个全回收Chunk的内存给释放了，以免出现频繁地申请)，如果是则将另一个全回收Chunk给Release了(释放在尾部的Chunk的全部堆上内存)。当检查到只有当前这一个Chunk全回收了，则通过swap可以很快地将这个Chunk交换到vector的末尾(vector的pop_back可以很快地删除元素 所以交换到末尾)，这样下次又出现全回收Chunk后可以直接检查vector的末尾是否全回收就能知道是否存在两个全回收Chunk。

void FixedAllocator::DoDeallocate(void* p)
{
    deallocChunk_->Deallocate(p,blockSize_);//调用下级Chunk的Deallocate，将回收的p区块作为p所属的这个deallocChunk_的first avaliable chunk第一个可用区块，下次allcoate时优先分配这个区块出去
    if(deallocChunk_->blocksAvaliable == numBlocks_)//此块Chunk的所有区块都回收回来了(当前可用区块数==一个Chunk的初始区块数)
    {
        Chunk& lastChunk = chunks_.back();//末尾Chunk
        if(&lastChunk==deallocChunk_)//当前全回收的chunk,就是chunks_向量的末尾的Chunk
        {
            //检查当前chunks_向量中是否有两个Chunk全回收了
            //这只需要往front方向看一眼，deallocChunk_[-1].blocksAvailable_--当前全回收Chunk的上个Chunk有多少个区块可用
            if(chunks_.size()>1 && deallocChunk_[-1].blocksAvailable_ == numBlocks_){
                //存在两个全回收Chunk 则将当前这个在向量末尾的Chunk内存给释放掉
                lastChunk.Release();
                chunks_.pop_back();
                allocChunk_ =deallocChunk_=&chunks_.front();
            }//deallocChunk_[-1].blocksAvailable_ == numBlocks_
            return;
        }//&lastChunk==deallocChunk_
        if(lastChunk.blocksAvaliable_==numBlocks_)
        {
            //存在两个全回收Chunk 抛弃最后一个
            lastChunk.Release();
            chunks_.pop_back();
            allocChunk_ = deallocChunk_;
        }
        else
        {
            //只有当前Chunk全回收了，则将当前全回收Chunk与chunks_的最后一块Chunk对调---及只要出现全回收Chunk则一定会将其放到chunks_末尾，这样每次查看是否有两个全回收Chunk，除了某个直接可判断的Chunk外，只需要再检查下chuks_.back() Chunk是否全回收就能知道是否有两个Chunk全回收了，可以将在末尾的全回收Chunk给释放掉
            std::swap(*deallocChunk_,lastChunk);
            allocChunk_=&chunks_.back();
        }
    }//blocksAvaliable == numBlocks_

}//函数结束

当同时出现两块全回收内存时会先释放emptyChunk_ 指向的这块(相当于上次出现的全回收Chunk)，然后将emptyChunk_ 交换到vector尾如果已经在尾部了就不管了，然后释放这个尾部Chunk的内存。最后将新出现的这个全回收Chunk作为赋值给emptyChunk_(下次释放先释放这个Chunk)。

void FixedAllocator::DoDeallocate(void* p)
{
    deallocChunk_->Deallocate(p,blockSize_);//调用下级Chunk的Deallocate
    if(deallocChunk_->blocksAvaliable == numBlocks_)
    {
        //当前deallocChunk_回收后 deallocChunk_为全回收状态
        if(NULL!=emptyChunk_)
        {
            //如果emptyChunk_已经指向了一个全回收chunk则我们需要将这个全回收chunk给释放掉
            Chunk* lastChunk = &chunks_.back();//末尾Chunk
            if( lastChunk == deallocChunk_)
            //chunks_末尾就是当前全回收的chunk,则将lastChunk 指向emptyChunk_这另外一个全回收Chunk，这样接下来lastChunk->Release(); 就是释放的这另一个全回收Chunk(emptyChunk_)
            	deallocChunk_ = emptyChunk_;
            else if( lastChunk != emptyChunk_)
            // 当前末尾不为记录的上个全回收Chunk,则把之前记录的全回收Chunk交换到chunks_向量的末尾,下面马上就会释放末尾Chunk,这样lastChunk->Release();释放的也是emptyChunk_
            	std::swap(*emptyChunk_,*lastChunk);

            assert(lastChunk->blocksAvaliable == numBlocks_);
            //释放末尾Chunk
            lastChunk->Release();
            chunks_.pop_back();
            //检查一下如果之前记录的allocChunk_为这个已经被释放的lastChunk，不改allocChunk_将会造成访问野内存，所以修改allocChunk_指向当前刚刚全回收的deallocChunk_，下次申请内存块优先从这取内存
            if(allocChunk_==lastChunk)
            	allocChunk_=deallocChunk_;
        }
        //标记全回收内存块指针 为当前这个刚刚检测到全回收的deallocChunk_，下次同时出现两块全回收会将emptyChunk_先释放
        emptyChunk_ = deallocChunk_;
    }
    assert((NULL==emptyChunk_)||(emptyChunk_->blocksAvaliable == numBlocks_));
}
