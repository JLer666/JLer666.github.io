def bubble_sort(arr):
    n = len(arr)
    for i in range(n):
        # 标记是否发生了交换
        swapped = False
        for j in range(0, n-i-1):
            if arr[j] > arr[j+1]:
                # 交换位置
                arr[j], arr[j+1] = arr[j+1], arr[j]
                swapped = True
        # 如果没有发生交换，说明列表已经有序，提前退出
        if not swapped:
            break
    return arr

//归并排序 O(nlogn)
// 归并排序的适用场景
// 需要稳定排序时
// 例如：排序对象包含多个属性，需保持原始顺序中的优先级。

// 数据规模较大且内存充足时
// 归并排序的 O(n) 额外空间在内存充足时不影响性能，适合大规模数据排序。

// 外部排序（如磁盘文件排序）
// 归并排序天然支持分块处理，适合无法一次性加载到内存的大数据。

// 链表排序
// 链表结构的随机访问成本高，归并排序的线性合并更适合链表。

// 并行计算
// 归并排序的分治过程天然可并行化，适合多核或分布式系统。

// 2. 快速排序的适用场景
// 内存敏感的场景
// 快速排序是原地排序，空间复杂度为 O(log n)，适合内存受限的环境（如嵌入式系统）。

class merge_sort {

    //merge是将两个有序数组合并为1个
    //归并在分割时的左右两个子数组一定是连续的所以可以直接将
    //n [l,mid]作为第一个数组 [mid+1,r]作为第二个数组
    void merge(vector<int> &n, int l, int mid, int r) {
        vector<int> temp(r - l + 1);//创建临时保存结果的空间
        int i = l;
        int j = mid + 1;
        int count = -1;
        //i,j分别指向两个子数组的头

        /**
         两个从大到小排序好的数组合并，
         就是双指针，初始化双指针分别指向各自的头
         看双指针指向的元素哪个更小，较小的那个肯定在合并后的数组中也是靠前的
         所以将较小的那个加入结果数组中，因为这个数已经用了 所以将对应的指针后移1位
         不断这样比较加入 知道其中一个数组完全遍历完
         **/
        while (i <= mid && j <= r) {//至少将其中一个子数组遍历完
            count++;
            if (n[i] <= n[j]) {
                temp[count] = n[i];
                i++;

            } else {
                temp[count] = n[j];
                j++;
            }
        }

        //看哪个数组有残留的直接拼到后面
        while (i <= mid)//数组1有残留
        {
            count++;
            temp[count] = n[i];
            i++;
        }
        while (j <= r)//数组2有残留
        {
            count++;
            temp[count] = n[j];
            j++;
        }
        //将临时结果数组合并后的结果 更新到原数组n中

        for (; count >= 0; count--) {
            n[l + count] = temp[count];
        }

    }

    void ms(vector<int> &n, int l, int r) {
        //1 分割 2 排序 3 合并 ---merge中的比较已经暗含了这个排序
        if (l >= r)return;
        int mid = l + (r - l) / 2;
        ms(n, l, mid);
        ms(n, mid + 1, r);
        merge(n, l, mid, r);
    }

三分搜索(ternary search)#

// 二分搜索的前提是在输入数据单调的前提下
// 三分搜索要求 被搜索的每个区间都是单峰函数(Unimodal Function)只存在唯一的极大值
// 单峰函数定义[l,r]存在极值点x*，[l,x*]上单调增或减。  [x*,r]上单调性与[l,x*]的相反
// 区间[A,B]存在f最大值，算法可用的前提是  存在x使得 所有满足A<=a<b<=x的a,b都有 f(a) < f(b)
//								              所有满足x<=a<b<=B的a,b都有 f(a) > f(b)
// we mean one of two behaviors of the function:
// The function strictly increases first, reaches a maximum (at a single point or over an interval), and then strictly decreases.
// The function strictly decreases first, reaches a minimum, and then strictly increases.

def ternary_search(f, left, right, absolute_precision) -> float:
    """Find maximum of unimodal function f() within [left, right].
    To find the minimum, reverse the if/else statement or reverse the comparison.
    """
    // l < m1 < m2 < r
    // if f(m1) < f(m2) 则最大值一定不在 左侧区间[l,m1] 在[m1,r]
    // if f(m1) > f(m2) 则最大值一定不在 右侧区间[m2,r] 在[l,m2]
    // if f(m1) == f(m2) 最大值在[m1,m2]
    while abs(right - left) >= absolute_precision:
        m1 = left + (right - left) / 3
        m2 = right - (right - left) / 3

        if f(m1) < f(m2): // f最大值在(m1,right]
            left = m1
        else:// f最大值在[left,m2)
            right = m2

    # Left and right are the current bounds; the maximum is between them
    return (left + right) / 2

double ternary_search_maximum(double l, double r) {
    double eps = 1e-9;              //set the error limit here
    while (r - l > eps) {
        double m1 = l + (r - l) / 3;
        double m2 = r - (r - l) / 3;
        double f1 = f(m1);      //evaluates the function at m1
        double f2 = f(m2);      //evaluates the function at m2
        if (f1 < f2)
            l = m1;
        else
            r = m2;
    }
    return f(l);                    //return the maximum of f(x) in [l, r]
}

int ternarySearch(int l, int r, int key, int ar[])
{
    while (r >= l) {

        // Find the mid1 and mid2
        int mid1 = l + (r - l) / 3;
        int mid2 = r - (r - l) / 3;

        // Check if key is present at any mid
        if (ar[mid1] == key) {
            return mid1;
        }
        if (ar[mid2] == key) {
            return mid2;
        }

        // Since key is not present at mid,
        // check in which region it is present
        // then repeat the Search operation
        // in that region

        if (key < ar[mid1]) {

            // The key lies in between l and mid1
            r = mid1 - 1;
        }
        else if (key > ar[mid2]) {

            // The key lies in between mid2 and r
            l = mid2 + 1;
        }
        else {

            // The key lies in between mid1 and mid2
            l = mid1 + 1;
            r = mid2 - 1;
        }
    }

    // Key not found
    return -1;
}

class SSP {
    // 单源图上最短路径

    int bfs(vector<vector<pair<int, int>>> &map) {
        // 解决 无向 无权 图 最短路径问题
        // 邻接表map[i]与i号节点 相连的其他节点的二维坐标 ,i号是二维坐标一维化后的 i = row*r + col
        queue<pair<int, int>> q;
        // 注意这里设我们的邻接表是方的 即每个节点的邻接节点数量都相等
        int r = map[0].size(), down = map.size();
        vector<vector<bool>> visited(down, vector<bool>(r, false));
        vector<vector<int>> cost(map.size(), vector<int>(map[0].size(), -1));//cost[i][j]--从规定的起点 走到 (i,j)所要花的最小代价
        cost[0][0] = 0;// 起点到起点的代价是0!!! --
        visited[0][0] = true;
        q.push({0, 0});
        // 得到(0,0) 到其他所有节点的 最短路径
        while (!q.empty()) {
            auto[row, col] = q.front();
            q.pop();// 在队列中的所有节点 从我们规定起点到这个点的坐标最短路径代价已经求出来了

            // 如果是要找期望终点的最短路径 这里再判断一下
//            if(row == target_row && col == target_col) return cost[row][col];

            // 注意这里设我们的邻接表是方的 即每个节点的邻接节点数量都相等
//            int r = map[0].size(), down = map.size();
            for (int j = 0; j < r; ++j) {
                // 根据题目的输入规定 这里直接得到邻接节点的坐标 不需要再判断坐标是否越界
                // BFS 遍历的节点深度 逐渐加深， 第一次碰到这个点的路径 一定是到达这个点的最短路径
                if (!visited[map[row * r + col][j].first][map[row * r + col][j].second]) {
                    visited[map[row * r + col][j].first][map[row * r + col][j].second] = true;
                    q.push(map[row * r + col][j]);
                    cost[map[row * r + col][j].first][map[row][j].second] = cost[row][col] + 1;// 无向无权图 单步代价就是1
                }
            }
        }
        return -1;
    }

    struct node {
    public:
        int x;
        int y;
        int cost;// 父节点到这个节点的代价(注意每个点的入度不一定为1 即不止一个父节点)
        node() : cost(-1) {}

        // 小顶堆
        bool operator<(const node &n2) const {
            return cost > n2.cost;
        }
    };
    // O((V+E)logV)
    int dijskstra(vector<vector<node>> &map) {
        // 解决 按最短路径长度不减(1->2的最短路径 一定比 1->2->n的最短路径短)的顺序求顶点间最短路径
        // dijskstra 不能有负权边！！！保证当前最小(第一次遇到--弹出的时候)就是它的最小值
        // 贪心依据 原点到每个点的最小路径只有两个可能，一是从原点到该点的边或是经过其他最短路径到达这个节点
        // 邻接表map[i]与i号节点 相连的其他节点的二维坐标 ,i号是二维坐标一维化后的 i = row*r + col 里面存的是对应节点到节点的权值

//        queue<pair<int, int>> q;
        priority_queue<node> q;
        // 注意这里设我们的邻接表是方的 即每个节点的邻接节点数量都相等
        int r = map[0].size(), down = map.size();
        vector<vector<bool>> visited(down, vector<bool>(r, false));
        vector<vector<node>> graph(map.size(), vector<node>(map[0].size()));
        graph[0][0].x = 0;
        graph[0][0].y = 0;
        graph[0][0].cost = 0;
        while (!q.empty()) {
            auto lcn = q.top();
            q.pop();// 在队列中的所有节点 从我们规定起点到这个点的坐标最短路径代价已经求出来了
            if (visited[lcn.x][lcn.y]) continue;// lcn已经在close中了，其最短路已经被确定过了 直接不看
            visited[lcn.x][lcn.y] = true;

            // 如果是要找期望终点的最短路径 这里再判断一下
//            if(lcn.x == target_row && lcn.y == target_col) return cost[lcn.x][lcn.y];

            // 注意这里设我们的邻接表是方的 即每个节点的邻接节点数量都相等
//            int r = map[0].size(), down = map.size();
            for (int j = 0; j < r; ++j) {
                // 根据题目的输入规定 这里直接得到邻接节点的坐标 不需要再判断坐标是否越界

                // (row,col)第j个邻居的cost = (row,col)cost + (row,col)到第j个邻居的边代价(map[row*r + col][j].cost)
                if (graph[map[lcn.x * r + lcn.y][j].x][map[lcn.x][j].y].cost == -1)// 第一次到达这个点
                {
                    graph[map[lcn.x * r + lcn.y][j].x][map[lcn.x][j].y].x = map[lcn.x * r + lcn.y][j].x;
                    graph[map[lcn.x * r + lcn.y][j].x][map[lcn.x][j].y].y = map[lcn.x * r + lcn.y][j].y;

                    graph[map[lcn.x * r + lcn.y][j].x][map[lcn.x][j].y].cost =
                            graph[lcn.x][lcn.y].cost + map[lcn.x * r + lcn.y][j].cost;
                    // 登记其父节点为lcn
                    //
                    q.emplace(graph[map[lcn.x * r + lcn.y][j].x][map[lcn.x][j].y]);
                } else if (graph[lcn.x][lcn.y].cost + map[lcn.x * r + lcn.y][j].cost <
                           graph[map[lcn.x * r + lcn.y][j].x][map[lcn.x][j].y].cost)// 不是第一次遇到了
                    // 从(row,col)第到其j个邻居的cost 小于之前保存的 到第j邻居cost
                {
                    // x,y之前已经存过了不需要存了 只需要修改cost为更小的那个cost
                    graph[map[lcn.x * r + lcn.y][j].x][map[lcn.x][j].y].cost =
                            graph[lcn.x][lcn.y].cost + map[lcn.x * r + lcn.y][j].cost;
                    // 修改其父节点为lcn
                    //

// 缓存的节点信息
// 优先淘汰访问频率最低元素 关心历史访问次数
// key_table键到节点映射   freq_table频率到对应频率到双向链表映射，同频率使用同一list按访问时间排序类似lru
struct Node {
int key, val, freq;
Node(int _key,int _val,int _freq): key(_key), val(_val), freq(_freq){}
};
class LFUCache {
int minfreq, capacity;
unordered_map<int, list<Node>::iterator> key_table;
unordered_map<int, list<Node>> freq_table;
public:
LFUCache(int _capacity) {
    minfreq = 0;
    capacity = _capacity;
    key_table.clear();
    freq_table.clear();
}

int get(int key) {
    if (capacity == 0) return -1;
    auto it = key_table.find(key);
    if (it == key_table.end()) return -1;
    list<Node>::iterator node = it -> second;
    int val = node -> val, freq = node -> freq;
    freq_table[freq].erase(node);
    // 如果当前链表为空，我们需要在哈希表中删除，且更新minFreq
    if (freq_table[freq].size() == 0) {
        freq_table.erase(freq);
        if (minfreq == freq) minfreq += 1;
    }
    // 插入到 freq + 1 中
    freq_table[freq + 1].push_front(Node(key, val, freq + 1));
    key_table[key] = freq_table[freq + 1].begin();
    return val;
}

void put(int key, int value) {
    if (capacity == 0) return;
    auto it = key_table.find(key);
    if (it == key_table.end()) {
        // 缓存已满，需要进行删除操作
        if (key_table.size() == capacity) {
            // 通过 minFreq 拿到 freq_table[minFreq] 链表的末尾节点
            auto it2 = freq_table[minfreq].back();
            key_table.erase(it2.key);
            freq_table[minfreq].pop_back();
            if (freq_table[minfreq].size() == 0) {
                freq_table.erase(minfreq);
            }
        }
        freq_table[1].push_front(Node(key, value, 1));
        key_table[key] = freq_table[1].begin();
        minfreq = 1;
    } else {
        // 与 get 操作基本一致，除了需要更新缓存的值
        list<Node>::iterator node = it -> second;
        int freq = node -> freq;
        freq_table[freq].erase(node);
        if (freq_table[freq].size() == 0) {
            freq_table.erase(freq);
            if (minfreq == freq) minfreq += 1;
        }
        freq_table[freq + 1].push_front(Node(key, value, freq + 1));
        key_table[key] = freq_table[freq + 1].begin();
    }
}
};

                    // 高度相同，总高+1
                    if (rank_[x_root] == rank_[y_root])

vector<Edge> kruskalMST(vector<Edge>& edges, int n) {
    sort(edges.begin(), edges.end(), [](const Edge& a, const Edge& b) {
        return a.weight < b.weight;  // 按权重升序排序
    });
    // 贪心优先将权重小的加入生成树中
    UnionFindSet uf(n);
    vector<Edge> mst;
    for (const Edge& e : edges) {
        // 树不允许存在环，通过并查集检查是否存在环
        // 按顺序将边加入并查集，当出现环后(当一条边的两个节点都属于同一集合则说明出现环)
        if (uf.Union(e.u, e.v)) {
            mst.push_back(e);
            if (mst.size() == n - 1) break;  // 已找到足够边
        }
    }
    return mst;
}

// ====================== Prim 算法实现 ======================
// 输入邻接表(稠密)
// 其原理本质是BFS泛洪用visited避免结果中存在环，则保证结果连通的树结构，并基于优先队列保证为最小权重树
vector<Edge> primMSTFullEdge(const vector<vector<pair<int, int>>>& adj) {
    int n = adj.size();
    vector<bool> visited(n, false);

    // 使用三元组存储：权重、源节点、目标节点
    using EdgeTuple = tuple<int, int, int>;
    priority_queue<EdgeTuple, vector<EdgeTuple>, greater<>> pq; // 最小堆

    vector<Edge> mst;
    int start = 0;
    visited[start] = true;

    // 初始化：将起始节点的所有邻接边加入队列
    for (const auto& [v, w] : adj[start]) {
        pq.emplace(w, start, v); // (weight, src, dest)
    }

    while (!pq.empty() && mst.size() < n - 1) {
        auto [weight, src, dest] = pq.top();
        pq.pop();

        if (visited[dest]) continue; // 目标节点已在生成树中

        // 记录完整边信息
        mst.emplace_back(src, dest, weight);
        visited[dest] = true;

        // 将目标节点的邻接边加入队列（源节点为当前目标节点）
        for (const auto& [v, w] : adj[dest]) {
            if (!visited[v]) {
                pq.emplace(w, dest, v); // (weight, src=dest, dest=v)
            }
        }
    }
    return mst;
}

// 节点为string
vector<string> topologicalSort(const unordered_map<string, unordered_set<string>>& edges) {
    // Step 1: 收集所有节点并初始化数据结构
    unordered_set<string> nodes;
    unordered_map<string, int> inDegree;

    // 收集所有节点并构建邻接表
    for (const auto& [u, neighbors] : edges) {
        nodes.insert(u);  // 添加源节点

        // 添加目标节点并初始化入度
        for (const auto& v : neighbors) {
            nodes.insert(v);
            inDegree[v] = 0;  // 确保所有节点都有入度记录
        }
    }

    // 初始化入度表（显式遍历保证覆盖孤立节点）
    for (const auto& node : nodes) {
        inDegree[node] = 0;  // 确保所有节点都有入度记录
    }

    // Step 2: 计算入度
    for (const auto& [u, neighbors] : edges) {
        for (const auto& v : neighbors) {
            inDegree[v]++;
        }
    }

    // Step 3: Kahn算法核心逻辑
    queue<string> q;
    for (const auto& node : nodes) {
        // 当v的入度==0相当于，v的所有前置条件已满足可加入结果中
        if (inDegree[node] == 0) q.push(node); // 入度为0，相当于优先级最高节点加入队列中
    }

    vector<string> order;
    while (!q.empty()) {
        string u = q.front();
        q.pop();
        order.push_back(u); // 加入结果

        // 处理邻接节点（需要检查是否存在邻接关系）
        if (edges.find(u) != edges.end()) {
            for (const auto& v : edges.at(u)) {
                if (--inDegree[v] == 0) { // 移除u->v边 -- v的入度-1.当v的入度==0相当于，v的所有前置条件已满足可加入结果中
                    q.push(v);
                }
            }
        }
    }

    // Step 4: 环检测
    return (order.size() == nodes.size()) ? order : vector<string>();
}

// 时间O(V+E)各个顶点和边只访问一次，空间O(V)
// 着色判断二部，如果某条边连通的两节点着相同的色则非二分图
// graph是邻接表表达
bool isBipartite(vector<vector<int>>& graph) {
    int n = graph.size();
    vector<int> color(n, -1); // -1: 未着色, 0/1: 两种颜色

    for (int i = 0; i < n; ++i) {
        if (color[i] != -1) continue;

        queue<int> q;
        q.push(i); // 找种子
        color[i] = 0;

        while (!q.empty()) {
            int u = q.front();
            q.pop();

            for (int v : graph[u]) {
                if (color[v] == -1) { // 未着色节点
                    color[v] = 1 - color[u]; // 连通节点染上另一种颜色
                    q.push(v);
                } else if (color[v] == color[u]) { // 连通节点的颜色相同不符合二分图定义
                    return false; // 冲突，非二分图
                }
            }
        }
    }
    return true;
}

            curr = curr->left;

        curr = curr->right; // 右
    }

vector<int> postorderTraversal(TreeNode* root) {
    vector<int> res;
    stack<TreeNode*> stk;
    TreeNode* curr = root;
    TreeNode* pre = nullptr; // 记录前一个访问节点

    while (curr != nullptr || !stk.empty()) {
        // 深入左子树到底
        while (curr) { // 左
            stk.push(curr);
            curr = curr->left;
        }

        TreeNode* node = stk.top();
        // 右子树存在且未被访问
        // 因为是左右中，是先遍历右子树，当回到中节点时需要pre检查判断右子树是否已被访问
        // 若已被访问不看右子树
        if (node->right != nullptr && node->right != pre) {
            curr = node->right; // 右
        } else { // 中
            res.push_back(node->val);
            pre = node; // 更新前驱节点
            stk.pop();
        }
    }
    return res;
}

// 判断两个字符串是否同构。同构的定义是，可以通过把一个字符串的某些相同的字符转换成
// 另一些相同的字符，使得两个字符串相同，且两种不同的字符不能够被转换成同一种字符。

bool isIsomorphic(string s, string t) {
    if(s.size() != t.size()) return false;
    vector<int> s_first_index(256, 0), t_first_index(256, 0);
    // 记录两个字符串每个位置的字符上一次出现的位置，如果两个字
    // 符串中相同位置的字符与它们上一次出现的位置一样，那么这两个字符串同构
    for (int i = 0; i < s.length(); ++i) {
        if (s_first_index[s[i]] != t_first_index[t[i]]) {
            return false;
        }
        s_first_index[s[i]] = t_first_index[t[i]] = i + 1;
    }
    return true;
}

int countSubstrings(string s) {

    for (int i = 0; i < s.length(); ++i) {

}

// Manacher 算法是在线性时间内求解最长回文子串的算法。
// Manacher 算法也会面临「方法一」中的奇数长度和偶数长度的问题，
// 它的处理方式是在所有的相邻字符中间插入 #，比如 abaa 会被处理成 #a#b#a#a#，
// 这样可以保证所有找到的回文串都是奇数长度的，以任意一个字符为回文中心，
// 既可以包含原来的奇数长度的情况，也可以包含原来偶数长度的情况
int countSubstrings(string s) {
    int n = s.size();
    string t = "$#";
    for (const char &c: s) {
        t += c;
        t += '#';
    }
    n = t.size();
    t += '!';

    // 以 t 的第 i 位为回文中心，可以拓展出的最大回文半径，
    // 那么 f(i)−1 就是以 i 为中心的最大回文串长度
    auto f = vector <int> (n);
    int iMax = 0, rMax = 0, ans = 0;
    for (int i = 1; i < n; ++i) { // 遍历待选的回文中心
        // 初始化 f[i]
        f[i] = (i <= rMax) ? min(rMax - i + 1, f[2 * iMax - i]) : 1;
        // 中心拓展
        while (t[i + f[i]] == t[i - f[i]]) ++f[i];
        // 动态维护 iMax 和 rMax
        if (i + f[i] - 1 > rMax) {
            iMax = i;
            rMax = i + f[i] - 1;
        }
        // 统计答案, 当前贡献为 (f[i] - 1) / 2 上取整
        ans += (f[i] / 2);
    }

    return ans;
}
// 696 01字符串，统计有多少非空01子串的0和1数量相同
int countBinarySubstrings(string s) {
    // 从左往右遍历数组，记录和当前位置数字相同且连续的长度，以及其之前连续的不同数字的长度
    // 将字符串 s 按照 0 和 1 的连续段分组，存在 counts 数组中，例如 s=00111011，
    // 以得到这样的 counts 数组：counts={2,3,1,2}
    // 这里 counts 数组中两个相邻的数一定代表的是两种不同的字符。
    // 假设 counts 数组中两个相邻的数字为 u 或者 v，它们对应着 u 个 0 和 v 个 1，
    // 或者 u 个 1 和 v 个 0。它们能组成的满足条件的子串数目为 min{u,v}，
    // 即一对相邻的数字对答案的贡献。
    // 对于某一个位置 i，其实我们只关心 i−1 位置的 counts 值是多少，
    // 所以可以用一个 last 变量来维护当前位置的前一个位置，
    // 这样可以省去一个 counts 数组的空间。
    int ptr = 0, n = s.size(), last = 0, ans = 0;
    while (ptr < n) {
        char c = s[ptr];
        int count = 0;
        while (ptr < n && s[ptr] == c) {
            ++ptr;
            ++count;
        }
        ans += min(count, last);
        last = count;
    }
    return ans;
}
// 227 给定含加减乘除运算的字符串，求结果只保留整数。没有()影响优先级
int calculate(string s) {
    vector<int> stk;
    char preSign = '+';
    int num = 0;
    int n = s.length();
    for (int i = 0; i < n; ++i) {
        if (isdigit(s[i])) {
            num = num * 10 + int(s[i] - '0'); // 左数字
        }

        if (!isdigit(s[i]) && s[i] != ' ' || i == n - 1) { // 符号或末端
            // 先进行所有乘除运算，并将这些乘除运算后的整数值放回原表达式的相应位置
            switch (preSign) {
                // 加减放入栈中
                case '+':
                    stk.push_back(num);
                    break;
                case '-':
                    stk.push_back(-num);
                    break;
                // 乘除直接进行运算
                case '*':
                    stk.back() *= num;
                    break;
                default:
                    stk.back() /= num;
            }
            preSign = s[i];
            num = 0;
        }
    }
    return accumulate(stk.begin(), stk.end(), 0);

// 28 字符串匹配
// 判断一个字符串是不是另一字符串的子字符串，是则返回其位置
// KMP算法 O(m+n)通过动规完成匹配

// 当出现不匹配时，找到不匹配字符之前的后缀，找到与这个后缀相同的前缀的后边开始继续匹配
// 如aabaaf，在f处不匹配，则找到f前有 最长的相同后缀(aa45)与前缀(aa12)
// 跳到前缀(aa12)的后一个字符b处开始继续匹配
// 在有前缀表后可以快速查询，在不匹配字符前的匹配子串的最长相等前后缀的长度
// aabaaf的前缀表是{0,1,0,1,2,0}
// 当在f处不匹配，其之前匹配字符aabaa的最长相等前后缀长为2.则直接跳到index=2处的b开始继续匹配

// O(m*n)的暴力匹配中。当needle(pattern)的某个字符不匹配时需要重新跳回needle的第一个字符重新开始匹配
// KMP解决的就是跳回重新头重新匹配浪费的问题。

// 可以将匹配过程理解为将needle(模式串)在haystack上滑动匹配的过程。
// 当needle出现不匹配时，直接将needle移动到haystack上下个已知的与needle前缀相同的位置上。
// 跳过重新对needle前缀的匹配。
// 这个needle前缀就是当前已完成匹配的needle[0:i]的最长的相等的真前缀与真后缀的长度对应的needle前缀

int strStr(string haystack, string needle) {
    int n = haystack.size(), m = needle.size();
    if (m == 0) {
        return 0;
    }
    vector<int> pi(m,0); // 模式串前缀表
    // pi[i] -- needle的子串needle[0:i]最长的相等的真前缀与真后缀的长度。
    // 特别地，如果不存在符合条件的前后缀，那么  pi[i]=0。
    // 其中真前缀与真后缀的定义为不等于自身的的前缀与后缀
    // 我们举个例子说明：字符串 aabaaab 的前缀函数值依次为 0,1,0,1,2,2,3。
    // π(0)=0，因为 a 没有真前缀和真后缀，根据规定为 0（可以发现对于任意字符串 π(0)=0 必定成立）；
    // π(1)=1，因为 aa 最长的一对相等的真前后缀为 a，长度为 1；
    // π(2)=0，因为 aab 没有对应真前缀和真后缀，根据规定为 0；
    // π(3)=1，因为 aaba 最长的一对相等的真前后缀为 a，长度为 1；
    // π(4)=2，因为 aabaa 最长的一对相等的真前后缀为 aa，长度为 2；
    // π(5)=2，因为 aabaaa 最长的一对相等的真前后缀为 aa，长度为 2；
    // π(6)=3，因为 aabaaab 最长的一对相等的真前后缀为 aab，长度为 3。
    for (int i = 1, j = 0; i < m; i++) {
        while (j > 0 && needle[i] != needle[j]) {
            j = pi[j - 1];  // 如果下一位不同，往前回溯
        }
        if (needle[i] == needle[j]) {
            j++;  // 如果下一位相同，更新相同的最大前缀和最大后缀长
        }
        pi[i] = j;
    }
    for (int i = 0, j = 0; i < n; i++) {
        while (j > 0 && haystack[i] != needle[j]) {
            j = pi[j - 1];  // 有部分匹配，往前回溯
        }
        if (haystack[i] == needle[j]) {
            j++;
        }
        if (j == m) {
            return i - m + 1; // 说明j移动到needle的最末端，返回相应的位置
        }
    }
    return -1;
}

// 单调栈模板（单调递减栈）
// 栈内元素始终保持单调递增或递减。常用于解决 下一个更大元素、下一个更小元素 等问题。
vector<int> nextGreaterElement(const vector<int>& nums) {
    int n = nums.size();
    vector<int> res(n, -1); // 默认值为-1
    stack<int> stk; // 单调递减栈

    for (int i = 0; i < n; ++i) {
        // 当栈不为空且当前元素大于栈顶元素时
        while (!stk.empty() && nums[i] > nums[stk.top()]) {
            res[stk.top()] = nums[i]; // 更新栈顶元素的下一个更大元素
            stk.pop(); // 弹出栈顶元素
        }
        stk.push(i); // 当前元素入栈
    }
    return res;
}

// 给定每日温度数组 T，返回到下个比当前高温日子需要等大的天数。
vector<int> dailyTemperatures(const vector<int>& T) {
    int n = T.size();
    vector<int> res(n, 0);
    stack<int> stk;

    for (int i = 0; i < n; ++i) {
        while (!stk.empty() && T[i] > T[stk.top()]) {
            int prevIdx = stk.top();
            res[prevIdx] = i - prevIdx; // 计算天数差
            stk.pop();
        }
        stk.push(i);
    }
    return res;
}

#include <vector>
#include <algorithm>
#include <stdexcept>

class MaxHeap {
private:
    vector<int> heap;

    // 上浮操作（插入时使用）
    void siftUp(size_t idx) {
        while (idx > 0) {
            size_t parentIdx = (idx - 1) / 2;
            if (heap[idx] <= heap[parentIdx]) break; // 满足大顶堆条件
            swap(heap[idx], heap[parentIdx]);
            idx = parentIdx;
        }
    }

    // 下沉操作（删除堆顶时使用）
    void siftDown(size_t idx) {
        size_t n = heap.size();
        while (true) {
            size_t leftChild = 2 * idx + 1;
            size_t rightChild = 2 * idx + 2;
            size_t largest = idx;

            // 找到当前节点、左子节点、右子节点中的最大值
            if (leftChild < n && heap[leftChild] > heap[largest]) {
                largest = leftChild;
            }
            if (rightChild < n && heap[rightChild] > heap[largest]) {
                largest = rightChild;
            }

            // 如果当前节点已经是最大值，则停止下沉
            if (largest == idx) break;

            swap(heap[idx], heap[largest]);
            idx = largest;
        }
    }

public:
    // 插入元素
    void push(int val) {
        heap.push_back(val); // 添加到末尾
        siftUp(heap.size() - 1); // 上浮调整
    }

    // 删除堆顶元素
    void pop() {
        if (heap.empty()) throw out_of_range("Heap is empty");
        heap[0] = heap.back(); // 用末尾元素替换堆顶
        heap.pop_back(); // 移除末尾元素
        if (!heap.empty()) siftDown(0); // 下沉调整
    }

    // 查看堆顶元素
    int top() const {
        if (heap.empty()) throw out_of_range("Heap is empty");
        return heap[0];
    }

    // 判断堆是否为空
    bool empty() const {
        return heap.empty();
    }

    // 返回堆的大小
    size_t size() const {
        return heap.size();
    }
};

template <typename T>
 class HashTable {
private:
    vector<list<T>> hash_table;
     // 哈希函数
    int myhash(const T & obj) const {
         return hash(obj, hash_table.size());
     }
 public:
    // size最好是质数
    HashTable(int size=31) {
         hash_table.reserve(size);
         hash_table.resize(size);
     }
    ~HashTable() {}
     // 查找哈希表是否存在该值
    bool contains(const T & obj) {
         int hash_value = myhash(obj);
         const list<T> & slot = hash_table[hash_value];
         std::list<T>::const_iterator it = slot.cbegin();
         for (; it != slot.cend() && *it != obj; ++it);
             return it != slot.cend();
     }
     // 插入值
    bool insert(const T & obj) {
         if (contains(obj)) {
             return false;
         }
         int hash_value = myhash(obj);
         std::list<T> & slot = hash_table[hash_value];
         slot.push_front(obj);
         return true;
     }
    // 删除值
    bool remove(const T & obj) {
     list<T> & slot = hash_table[myhash(obj)];
     auto it = find(slot.begin(), slot.end(), obj);
     if (it == slot.end()) {
         return false;
     }
     slot.erase(it);
     return true;
     }
 };
