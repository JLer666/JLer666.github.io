allocator 例子

提前申请一大块内存(减少malloc的次数 加快程序运行)，使用者在new时再从这个大块内存中分配空间给对象，其实还有一个好处就是malloc分配内存会创建出多余的cookie(8个字节)，我们自己分配就可以不去创建cookie从而节省下一些空间。

利用内存池的观念(即创建出一大段连续空间的内存，然后将其切割成一小段一小段)，将创建的元素对象放在内存池切分好的各分段小内存片中，这样避免了多次调用new而造成生成多个带有cookie的内存空间。可以生成一大段只带有两个头尾cookie的内存空间，而该一大段内存空间又被切分成每一小段的内存空间，且其中的每一小段内存空间片都可以共享这一整体的cookie信息。

1per-class allocator (c++ primer p765)

#include <cstddef>
#include <iostream>
using namespace std;
//这种内存针对单一类的内存管理角坐 per-class allocator
class Screen
{
public:
    Screen(int x) : i(x){};
    int get() { return i; }
    static void *operator new(size_t);           //重载op new  // 其实这里最规范的写法应该添加static关键字！！
    static void operator delete(void *, size_t); //重载op delete() // 虽然不添加系统也会默认将其看成 static函数去调用！！
    //delete expression只调用一个版本,delete(void*).其它的版本只在出现异常时调用.
    //为什么不定义operator delete(void *)?
    static void operator delete(void *);
    //我们并没有重载void *operator new(size_t); 对应的delete void operator delete(void *)
    //...
private:
    Screen *next;             //这种设计会多耗一个next
    static Screen *freeStore; // 静态变量必须类内声明，类外定义赋初值
    static const int screenChunk;
    int i;
};
Screen *Screen::freeStore = 0;      //余下未分配空间的首地址
const int Screen::screenChunk = 24; //1块内存大24个sizeof(Screen)大
void *Screen::operator new(size_t size)
{
    Screen *p;
    if (!freeStore) //每当没有余下空间后 重新申请一块大内存
    {
        //linked list是空的直接申请一大块 一次申请24个size的空间
        size_t chunk = screenChunk * size;
        freeStore = p = reinterpret_cast<Screen *>(new char[chunk]);
        //将这一大块内存 串起来变为linked list
        //p初始指向分配空间的首地址空间共size个Screen大
        //将每个Screen像链表一样连接起来
        for (; p != &freeStore[screenChunk - 1]; ++p)
            p->next = p + 1;
        p->next = 0; //链表尾指向0
    }
    p = freeStore;               //p重新指向未分配空间首地址
    freeStore = freeStore->next; //freeStore指向链表头指向的下一个Screen对象
    //p(链表头head)->head_next->....
    //                / \
    //                 |
    //             freeStore
    return p; //传回一块(链表头的那块) freeStore指向余下的空间(第一次23块)
}
//delete expression会先调用析构函数 再调用op delete回收空间
//本质将传进来的 要被回收的p空间(一个Screen节点) 插到余下未分配链表！！！的头  这样就回收完成了 下次再new时就会优先分配这个空间
void Screen::operator delete(void *p, size_t size)和
void Screen::operator delete(void *p)
在类中重写 只能选择其中一种 同时写这两个函数 编译器会报错 但是GNUC这两个函数的全局版本可以共存 但是还是建议只写一种
void Screen::operator delete(void *p, size_t size)
{
    //将传入的回收空间p点 插到链表头(对于单向链表来说 插入操作 插到链表头是最快的)
    //将已经被删除的对象插回free list前端
    (static_cast<Screen *>(p))->next = freeStore; //
    freeStore = static_cast<Screen *>(p);         //余下可分配空间多出一个p界点
}
//delete expression只调用一个版本,delete(void*).其它的版本只在出现异常时调用.
//为什么不定义operator delete(void *)?
void Screen::operator delete(void *p)
{
    //将传入的回收空间p点 插到链表头(对于单向链表来说 插入操作 插到链表头是最快的)
    //将已经被删除的对象插回free list前端
    (static_cast<Screen *>(p))->next = freeStore; //
    freeStore = static_cast<Screen *>(p);         //余下可分配空间多出一个p界点
}
//注意freeStore是静态对象 只要有Screen对象delete就会将空间回收到freeStore
//freeStore链表的大小会越来越大
//链表的最终大小是总共申请过的Screen数量 + (小于 24 * sizeof(Screen) 的冗余大小)
//注意这个op delete在没出错的情况下并没有free or op delete 将申请来的内存归还给OS

2per-class allocator union(effective C++ item10)

//effective c++ item10
class Airplane
{
private:
    struct AirplaneRep //Airplane represent 表达内容 将Airplane对象的属性成员变量保存 规划到一起
    {
        unsigned long miles; //4字节
        char type;           //1字节
    };                       //共5字节 但是编译器会做对齐 将会变为8字节

    union //嵌入式指针
    {
        AirplaneRep rep; //此rep针对使用中的对向  c++可以不用加struct这个声明符
        Airplane *next;  //此指针 针对 内存块list上的对象
    };                   //rep和next共用同一块空间 AirplaneRep 5字节 Airplane * 4字节
    //union回取其中空间最大的即 5字节作为这个联合体的大小
    //但是编译器会做对齐 将会变为8字节
    //next 将联合体(8字节)的前4字节视为一个指针
    //注意我们并没有对这个union命名，(命名了就可以将名称作为一种type代表这个联合体类型)
    //未加名称表示这是一个anonymous union匿名联合体
    //其内部的变量可以直接视为是在类内定义的成员变量！！！但是这些成员变量共用一块空间！！！
    //用union的话我们更改rep内容会不会影响next指针的指向?
    //不需要考虑这个问题 next只有在申请(申请一大块的时候)和回收内存时有用到
    //，当我们申请到一个对象的内存 此时这个对象以经脱离链表了其next已经没有意义了
    //我们只在管理内存的时候要用到next!!!!当我们在用对象属性做其他事的时候使用Rep 不用管next的死活
    //所以next这部分空间我们可以随意修改  这么做省下next的四字节
public:
    unsigned long getMiles() { return rep.miles; }
    char getType() { return rep.type; }
    void set(unsigned long m, char t)
    {
        rep.miles = m;
        rep.type = t;
    }
    static void *operator new(size_t size);
    static void operator delete(void *, size_t size);
private:
    static const int BLOCK_SIZE;
    static Airplane *head_of_FreeList;
};
const int Airplane::BLOCK_SIZE = 512;
Airplane *Airplane::head_of_FreeList = 0;
void *Airplane::operator new(size_t size)
{
    //如果传入的size大小有误 转交给全局版的op new处理
    if (size != sizeof(Airplane)) //在继承发生的时候会出现这个现象
        return ::operator new(size);
    Airplane *p = head_of_FreeList;
    if (p)
        head_of_FreeList = p->next;
    else
    {
        //p为null(0) 表示head_of_FreeList指向0 为链表尾 链表上已经没有空余内存节点分配 需要分配新的内存块
        Airplane *newBlock = static_cast<Airplane *>(::operator new(BLOCK_SIZE * sizeof(Airplane))); //一次申请512个Airplane那么大
        //上个版本是用expression new(先调用构造 再调用op new)分配空间  这里用op new来分配空间
        //这512块只用了头尾两个cookie 8字节
        //将每一块串成list
        //但是会跳过第0块 因为其会作为本次申请内存的结果 返回
        for (int i = 1; i < BLOCK_SIZE - 1; i++)
            newBlock[i].next = &newBlock[i + 1]; //[i+1]已经被创造出来了 所以其地址存在 没问题
        newBlock[BLOCK_SIZE - 1].next = 0;       //最后链表尾指向0
        p = newBlock;
        head_of_FreeList = &newBlock[1]; //因为0(newBlock)是此次被分配的内存 所以freelist的头指向的是1
    }
    return p;
}
//delete expression会先调用析构函数 再调用op delete回收空间
//本质将传进来的 要被回收的p空间(一个Screen节点) 插到余下未分配链表！！！的头  这样就回收完成了 下次再new时就会优先分配这个空间
void Airplane::operator delete(void *deadObj, size_t size)
{
    if (deadObj == 0)
        return;
    if (size != sizeof(Airplane))
    {
        ::operator delete(deadObj);
        return;
    }
    //将传入的回收空间p点 插到链表头(对于单向链表来说 插入操作 插到链表头是最快的)
    //将已经被删除的对象插回free list前端
    // (static_cast<Airplane *>(deadObj))->next = head_of_FreeList; //
    // head_of_FreeList = static_cast<Airplane *>(deadObj);         //余下可分配空间多出一个p界点
    //收回来的内存放在链表的头
    Airplane *carcass = static_cast<Airplane *>(deadObj);
    carcass->next = head_of_FreeList;
    head_of_FreeList = carcass;
}
//注意head_of_FreeList是静态对象 只要有Airplane对象delete就会将空间回收到head_of_FreeList
//head_of_FreeList链表的大小会越来越大
//链表的最终大小是总共申请过的Airplane数量 + (小于 512 * sizeof(Airplane) 的冗余大小)
//注意这个op delete在没出错的情况下并没有free or op delete 将申请来的内存归还给OS
//delete expression只调用一个版本,delete(void*).其它的版本只在出现异常时调用.
//为什么不定义operator delete(void *)?

3static allocator

#include <cstddef>
#include <iostream>
using namespace std;
//class allocator 很好 但是是在类内定义的只适用于 那个类
//对于不同类来说这样的allocator的op new和op delete做的动作都几乎相同
//所以做一个单独类专门做内存管理
//不同类可以在其内部定义一个allocator 专门用于那个类的内存管理
class allocator_ //叫allocator会与全局版的冲突  本质上就是在管理一个链表
{
private:
    struct obj
    {
        obj *next; //嵌入式指针
    };
public:
    void *allocate(size_t);
    void deallocate(void *, size_t);
private:
    obj *freeStore = nullptr; //就是一个指针 4字节 用于存储我们真正要存的对象的地址
    const int CHUNK = 5;
};
// 这个allocator_有个局限 allocator_维护的一条链表每次扩充的空间都是固定只有CHUNK
// 在std allocator中则有16条链表 用于应付不同的CHUNK！！！！
void allocator_::deallocate(void *p, size_t)
{
    //将传入的回收空间p点 插到链表头(对于单向链表来说 插入操作 插到链表头是最快的)
    //将已经被删除的对象插回free list前端
    (static_cast<obj *>(p))->next = freeStore; //
    freeStore = static_cast<obj *>(p);         //余下可分配空间多出一个p界点
}
void *allocator_::allocate(size_t size)
{
    obj *p;
    if (!freeStore) //当前链表已经没有新的空间了 则分配一大块空间并串起来
    {
        //p为null(0) 表示head_of_FreeList指向0 为链表尾 链表上已经没有空余内存节点分配 需要分配新的内存块
        size_t chunk = CHUNK * size; //一次分配5个要创建对象的大小
        freeStore = p = (obj *)malloc(chunk);
        //注意 我们只知道 我们要分配的单个对象的大小 但是不知道我们要分配对象的类型
        //所以这里malloc没办法强转为那个对象的类型  但是知道大小一样也可以将这一大块空间
        //分割为多个 szieof(对象)的小块并用指针(上个小块的next指针)指向这小块的首地址 串成一个链表
        // p指针是在不断移动的 不断地指向每一个size大小内存块的首地址
        //(p其实指向的是size大小空间的前四个字节组成的obj的首地址)
        // |next指针 .... 这块有size大|...next指针 .... 这块有size大|....|
        // 每块空间首地址被强转为obj型 所以空间中的前4字节暂时被当作next指针 串联其他size大小的空间
        //      |-------------------------|(next指向下一块的首地址)
        // 只有在分配和回收的时候我们才需要这个next指针
        // 在allocate return了这块空间的首地址后 内存块已经不在链表里了next指针没有意义了
        // 这四个字节可以去存真正的对象大小
        // 将每一块串成list
        for (int i = 0; i < CHUNK - 1; i++)
        {
            p->next = (obj *)((char *)p + size); //将空间强转为obj型
            //size为被管理类 的一个对象的大小！！！！ 单位字节 所以要将p转为char型 也变为以字节为单位
            p = p->next;
        }
        p->next = nullptr; //last
    }
    p = freeStore; //将当前可用空间的头切出来返回
    freeStore = freeStore->next;
    return p;
}
//其他使用这个allocator的类通过一个静态成员变量来使用这个分配器
class Foo
{
public:
    long L;
    string str;
    static allocator_ myAlloc; //静态的内存管理器 专门管理这个类的所有对象
public:
    Foo(long l) : L(l) {}
    static void *operator new(size_t size)
    {
        return myAlloc.allocate(size);
    }
    static void operator delete(void *pdead, size_t size)
    {
        return myAlloc.deallocate(pdead, size);
    }
};
allocator_ Foo::myAlloc;

namespace jj10
{
    using ::allocator_; //声明我们在全局命名空间  定义的分配器 这样我们在这个命名空间内才可以用
    // DECLARE_POOL_ALLOC -- used in class definition
#define DECLARE_POOL_ALLOC()                                           \
public:                                                                \
    void *operator new(size_t size) { return myAlloc.allocate(size); } \
    void operator delete(void *p) { myAlloc.deallocate(p, 0); }        \
                                                                       \
protected:                                                             \
    static allocator_ myAlloc;
// DECLARE_POOL_ALLOC()这个宏代表了上面这么长的一串代码  \是宏的换行符
// IMPLEMENT_POOL_ALLOC -- used in class implementation file
// 这是个代参宏 将类名传入 然后这个宏直接可替换为 allocator 类名::myAlloc; 这句话
#define IMPLEMENT_POOL_ALLOC(class_name) \
    allocator_ class_name::myAlloc;
    // 使用了宏替换代码 类定义变的非常简洁了
    // in class definition file
    class Foo
    {
        DECLARE_POOL_ALLOC()
    public:
        long L;
        string str;
    public:
        Foo(long l) : L(l) {}
    };
    //in class implementation file
    IMPLEMENT_POOL_ALLOC(Foo)
} //namespace
