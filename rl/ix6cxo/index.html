下面是不可取的直接计算KL散度二阶导数的方法

 def hessian(network, states):
    #pa = network.forward(states)
    pa_sum = network.get_kl(states)
    # pa_sum--Dkl  J--dDkl/dθ
    J = torch.autograd.grad(pa_sum, network.parameters(), create_graph=True, retain_graph=True)
    J_ = Tensor().cuda()

# concatenate the various gradient tensors (for each layer) into one vector

# 将J--dDkl/dθ展开打平
    for grad in J:
        J_ = torch.cat((J_, grad.view(-1)), 0)

    H = Tensor().cuda()

# calculate gradient wrt each element and concatenate into the Hessian matrix

# JJ = d2Dkl/dθ^2
    for Ji in J_:
        JJ = torch.autograd.grad(Ji, network.parameters(), create_graph=False, retain_graph=True)
        JJ_ = torch.cat([grad.contiguous().view(-1) for grad in JJ])
        H = torch.cat((H, JJ_), 0)
    # numParams is the number of parameters in the network
    numParams = sum(p.numel() for p in network.parameters() if p.requires_grad)
    HH = H.view((numParams, numParams))
    return HH

这里使用一个技巧来避免在求H-1g(**)的过程中直接去求KL散度的二阶导,**直接求的公式 $ H_{ij}=\frac{\partial}{\partial \theta_j}\frac{\partial D_{KL}}{\partial \theta_i} $

取H中的一行[1,K]*s[K,1] = 1 是个scalar。取H的第k行乘上s向量得到的结果如下

$ y_k = \sum_jH_{kj}s_j=\sum_j\frac{\partial}{\partial \theta_j}\frac{\partial D_{KL}}{\partial \theta_k}s_j=\frac{\partial}{\partial \theta_k}\sum_j\frac{\partial D_{KL}}{\partial \theta_j}s_j $

为了求Hs [K,1]的第k行元素,可以先计算KL散度的一阶导 再将一阶导和输入的s向量相乘$ \frac{\partial D_{KL}}{\partial \theta_j}_{1\times 1}{s_j}_{1\times 1} == _{1\times 1} $得到的是个scalar,再计算这个scalar的导数。

这种方法是不借助FIM阵的直接计算方法 叫直接方法

def Fvp_direct(network, states, v):
    damping = 1e-2
    #pa = network.forward(states)
    # pa_sum--Dkl
	pa_sum = network.get_kl(states)
	# dDkl/dθ
	grads = torch.autograd.grad(pa_sum, network.parameters(), create_graph=True)
    grads_flat = torch.cat([grad.view(-1) for grad in grads])
    # compute the dot product with the input vector
    # grads_v = dDkl/dθ * s
    grads_v = torch.sum(grads_flat * v)
    # now compute the derivative again.
    # grads_grads_v = d(dDkl/dθ * s)/dθ
    grads_grads_v = torch.autograd.grad(grads_v, network.parameters(), create_graph=False)
    flat_grad_grad_v = torch.cat([grad.contiguous().view(-1) for grad in grads_grads_v]).data
    当v太大后对Dkl的二次展开的置信度降低，因此取B=H+/\I，/\(damping )被称为阻尼系数，Bv=Hv+/\v表示费舍尔向量积(Fisher Vector Product)。
    return flat_grad_grad_v + v * damping

我们需要的是$ \triangledown_{\theta}^2 D_{KL}^{s\sim\rho_{\theta_{old}}}(\pi_{\theta_{old}}||\pi_{\theta})_{\theta=\theta_{old}} $,即在θ_old处的二阶导

$ \triangledown_{\theta} D_{KL}^{s\sim\rho_{\theta_{old}}}(\pi_{\theta_{old}}||\pi_{\theta})=\frac{\partial\pi_{\theta}}{\partial\theta}\frac{\partial D_{KL}^{s\sim\rho_{\theta_{old}}}(\pi_{\theta_{old}}||\pi_{\theta})}{\partial \pi_{\theta}} $

其中dpi/dθ[K,M] dDkl/dpi[M,1] 所以一阶导结果是[K,1]

$ \triangledown_{\theta}^2 D_{KL}^{s\sim\rho_{\theta_{old}}}(\pi_{\theta_{old}}||\pi_{\theta})=\frac{\partial^2\pi_{\theta}}{\partial\theta^2}\frac{\partial D_{KL}^{s\sim\rho_{\theta_{old}}}(\pi_{\theta_{old}}||\pi_{\theta})}{\partial \pi_{\theta}} + \frac{\partial\pi_{\theta}}{\partial\theta}\frac{\partial\frac{\partial D_{KL}^{s\sim\rho_{\theta_{old}}}(\pi_{\theta_{old}}||\pi_{\theta})}{\partial \pi_{\theta}}}{\partial \theta} $其中第一项

$ \begin{gathered} \because \frac{\partial D_{KL}^{s\sim\rho_{\theta_{old}}}(\pi_{\theta_{old}}||\pi_{\theta})}{\partial \theta}_{\theta = \theta_{old}}=0\\ \therefore \frac{\partial^2\pi_{\theta}}{\partial\theta^2}\frac{\partial D_{KL}^{s\sim\rho_{\theta_{old}}}(\pi_{\theta_{old}}||\pi_{\theta})}{\partial \pi_{\theta}}= \frac{\partial\pi_{\theta}}{\partial\theta}\frac{\partial D_{KL}^{s\sim\rho_{\theta_{old}}}(\pi_{\theta_{old}}||\pi_{\theta})}{\partial \theta}=0 \end{gathered} $可以这么理解当θ=θ_old时KL散度等于0为全局最小值所以KL散度在那点对θ的导数一定是0(但注意 一阶导为0的点 二阶导不一定为0)?

$ \frac{\partial\pi_{\theta}}{\partial\theta}\frac{\partial\frac{\partial D_{KL}^{s\sim\rho_{\theta_{old}}}(\pi_{\theta_{old}}||\pi_{\theta})}{\partial \pi_{\theta}}}{\partial \theta}= \frac{\partial\pi_{\theta}}{\partial\theta} \frac{\partial^2 D_{KL}^{s\sim\rho_{\theta_{old}}}(\pi_{\theta_{old}}||\pi_{\theta})}{\partial \pi_{\theta}^2} (\frac{\partial\pi_{\theta}}{\partial\theta})^T $

其中dpi/dθ[K,M] d2Dkl/dpi2[M,M] 所以结果是 [K,M] [M,M] [M,K] = [K,K]

$ \begin{gathered} D_{KL}^{s\sim\rho_{\theta_{old}}}(\pi_{\theta_{old}}||\pi_{\theta})=D_{KL}^{s\sim\rho_{\theta_{old}}}(\pi_{\theta_{old}}(\cdot|s)||\pi_{\theta}(\cdot|s))\\=\sum_{a_i}\pi_{\theta_{old}}(a=a_i|s)\cdot ln\frac{\pi_{\theta_{old}}(a=a_i|s)}{\pi_{\theta}(a=a_i|s)} \end{gathered} $

将pi视为一个函数

$ =\sum_{a_i}\pi_{\theta_{old}}(a_i,s)\cdot ln\frac{\pi_{\theta_{old}}(a_i,s)}{\pi_{\theta}(a_i,s)} $

$ \begin{gathered} \frac{\partial D_{KL}^{s\sim\rho_{\theta_{old}}}(\pi_{\theta_{old}}(\cdot|s)||\pi_{\theta}(\cdot|s))}{\partial \pi_{\theta}(\cdot|s)}\\=\frac{\partial \sum_{a_i;i=1,..,}\pi_{\theta_{old}}(a_i,s)\cdot ln\frac{\pi_{\theta_{old}}(a_i,s)}{\pi_{\theta}(a_i,s)}}{\partial \pi_{\theta}(\cdot|s)}\\= -\frac{\partial \sum_{a_i;i=1,..,}\pi_{\theta_{old}}(a_i,s)\cdot ln\pi_{\theta}(a_i,s)}{\partial \pi_{\theta}(\cdot|s)} \\= -\frac{\partial \sum_{a_i;i=1,..,}\pi_{\theta_{old}}(a_i,s)\cdot ln\pi_{\theta}(a_i,s)}{\partial [\pi_{\theta}(a_1|s),...,\pi_{\theta}(a_M|s)]^T} \\= [-\frac{\partial \sum_{a_i;i=1,..,}\pi_{\theta_{old}}(a_i,s)\cdot ln\pi_{\theta}(a_i,s)}{\partial \pi_{\theta}(a_1|s)},...,-\frac{\partial \sum_{a_i;i=1,..,}\pi_{\theta_{old}}(a_i,s)\cdot ln\pi_{\theta}(a_i,s)}{\partial \pi_{\theta}(a_M|s)}]^T \\ = [-\frac{\pi_{\theta_{old}}(a_i,s)}{\pi_{\theta}(a_1|s)},...,-\frac{\pi_{\theta_{old}}(a_M,s)}{\pi_{\theta}(a_M|s)}]^T ;M=action_num,\pi(s)_{(M\times 1)} \end{gathered} $

pi old和pi都是动作分布 [M,1]

$ \frac{\partial^2 D_{KL}^{s\sim\rho_{\theta_{old}}}(\pi_{\theta_{old}}||\pi_{\theta})}{\partial \pi_{\theta}^2}=\frac{\partial[-\frac{\pi_{\theta_{old}}(a_1,s)}{\pi_{\theta}(a_1,s)},...,-\frac{\pi_{\theta_{old}}(a_M,s)}{\pi_{\theta}(a_M,s)}]^T}{\partial\pi_{\theta} }=\frac{\partial[-\frac{\pi_{\theta_{old}}(a_1,s)}{\pi_{\theta}(a_1,s)},...,-\frac{\pi_{\theta_{old}}(a_M,s)}{\pi_{\theta}(a_M,s)}]^T}{\partial[\pi_{\theta}(a_1,s),...,\pi_{\theta}(a_M,s)]^T } $

$ =\begin{bmatrix} \frac{\partial-\frac{\pi_{\theta_{old}}(a_1,s)}{\pi_{\theta}(a_1,s)} }{\partial\pi_{\theta}(a_1,s)}& ... & \frac{\partial-\frac{\pi_{\theta_{old}}(a_1,s)}{\pi_{\theta}(a_1,s)} }{\partial\pi_{\theta}(a_M,s)} \\ \vdots& \ddots & \\ \frac{\partial-\frac{\pi_{\theta_{old}}(a_M,s)}{\pi_{\theta}(a_M,s)} }{\partial\pi_{\theta}(a_1,s)} & & \frac{\partial-\frac{\pi_{\theta_{old}}(a_M,s)}{\pi_{\theta}(a_M,s)} }{\partial\pi_{\theta}(a_M,s)}\\ \end{bmatrix} $

$ =\begin{bmatrix} \frac{\pi_{\theta_{old}}(a_1,s)}{\pi_{\theta}^2(a_1,s)}& 0... & 0\\ \vdots& \ddots & \\ 0& & \frac{\pi_{\theta_{old}}(a_M,s)}{\pi_{\theta}^2(a_M,s)}\\ \end{bmatrix} $

$ \frac{\partial^2 D_{KL}^{s\sim\rho_{\theta_{old}}}(\pi_{\theta_{old}}||\pi_{\theta})}{\partial \pi_{\theta}^2}_{\theta=\theta_{old}} $

$ =\begin{bmatrix} \frac{\pi_{\theta_{old}}(a_1,s)}{\pi_{\theta}^2(a_1,s)}& 0... & 0\\ \vdots& \ddots & \\ 0& & \frac{\pi_{\theta_{old}}(a_M,s)}{\pi_{\theta}^2(a_M,s)}\\ \end{bmatrix}_{\theta=\theta_{old}} $

$ =\begin{bmatrix} \frac{1}{\pi_{\theta_{old}}(a_1,s)}& 0... & 0\\ \vdots& \ddots & \\ 0& & \frac{1}{\pi_{\theta_{old}}(a_M,s)}\\ \end{bmatrix} $

$ \frac{\partial \pi_{\theta}}{\partial \theta}=\frac{\partial[\pi_{\theta}(a_1,s),...,\pi_{\theta}(a_M,s)]^T}{\partial[\theta_1,...,\theta_k]^T} \\ =\begin{bmatrix} \frac{\partial\pi_{\theta}(a_1,s)}{\partial \theta_1}& ... & \frac{\partial\pi_{\theta}(a_M,s)}{\partial \theta_1}\\ \vdots& \ddots & \\ \frac{\partial\pi_{\theta}(a_1,s)}{\partial \theta_k}& & \frac{\partial\pi_{\theta}(a_M,s)}{\partial \theta_k}\\ \end{bmatrix} $

$ Hv=\triangledown_{\theta}^2 D_{KL}^{s\sim\rho_{\theta_{old}}}(\pi_{\theta_{old}}||\pi_{\theta})\cdot v= \frac{\partial \pi_{\theta}}{\partial \theta}\frac{\partial^2 D_{KL}^{s\sim\rho_{\theta_{old}}}(\pi_{\theta_{old}}||\pi_{\theta})}{\partial \pi_{\theta}^2}(\frac{\partial \pi_{\theta}}{\partial \theta})^T\cdot v $

$ =\begin{bmatrix} \frac{\partial\pi_{\theta}(a_1,s)}{\partial \theta_1}& ... & \frac{\partial\pi_{\theta}(a_M,s)}{\partial \theta_1}\\ \vdots& \ddots & \\ \frac{\partial\pi_{\theta}(a_1,s)}{\partial \theta_k}& & \frac{\partial\pi_{\theta}(a_M,s)}{\partial \theta_k}\\ \end{bmatrix} \cdot \begin{bmatrix} \frac{1}{\pi_{\theta_{old}}(a_1,s)}& 0... & 0\\ \vdots& \ddots & \\ 0& & \frac{1}{\pi_{\theta_{old}}(a_M,s)}\\ \end{bmatrix} \cdot \begin{bmatrix} \frac{\partial\pi_{\theta}(a_1,s)}{\partial \theta_1}& ... & \frac{\partial\pi_{\theta}(a_1,s)}{\partial \theta_k}\\ \vdots& \ddots & \\ \frac{\partial\pi_{\theta}(a_M,s)}{\partial \theta_1}& & \frac{\partial\pi_{\theta}(a_M,s)}{\partial \theta_k}\\ \end{bmatrix}\cdot \begin{bmatrix} v_1\\ \vdots\\ v_k\\ \end{bmatrix} $

先看

$ \begin{bmatrix} \frac{\partial\pi_{\theta}(a_1,s)}{\partial \theta_1}& ... & \frac{\partial\pi_{\theta}(a_1,s)}{\partial \theta_k}\\ \vdots& \ddots & \\ \frac{\partial\pi_{\theta}(a_M,s)}{\partial \theta_1}& & \frac{\partial\pi_{\theta}(a_M,s)}{\partial \theta_k}\\ \end{bmatrix}\cdot \begin{bmatrix} v_1\\ \vdots\\ v_k\\ \end{bmatrix}= \begin{bmatrix} \frac{\partial\pi_{\theta}(a_1,s)}{\partial \theta_1}v_1+\frac{\partial\pi_{\theta}(a_1,s)}{\partial \theta_2}v_2+...+\frac{\partial\pi_{\theta}(a_1,s)}{\partial \theta_k}v_k\\ \frac{\partial\pi_{\theta}(a_2,s)}{\partial \theta_1}v_1 + \frac{\partial\pi_{\theta}(a_2,s)}{\partial \theta_2}v_2+... + \frac{\partial\pi_{\theta}(a_2,s)}{\partial \theta_k}v_k\\ \vdots \\ \frac{\partial\pi_{\theta}(a_M,s)}{\partial \theta_1}v_1 + \frac{\partial\pi_{\theta}(a_M,s)}{\partial \theta_2}v_2+...+ \frac{\partial\pi_{\theta}(a_M,s)}{\partial \theta_k}v_k \\ \end{bmatrix} $

为了求得这个形式的矩阵，构造$ J_t=\sum_{i=1}^M \pi_{\theta}(a_i,s)t_i $，向量[t1,..tm]为随意的一个常数向量 与参数θ无关。

计算

$ \frac{\partial J_t}{\partial \theta}=\sum_{i=1}^M \pi_{\theta}(a_i,s)t_i= t_1[\frac{\partial\pi_{\theta}(a_1,s)}{\partial \theta_1},\frac{\partial\pi_{\theta}(a_1,s)}{\partial \theta_2},...,\frac{\partial\pi_{\theta}(a_1,s)}{\partial \theta_k}]+ t_2[\frac{\partial\pi_{\theta}(a_2,s)}{\partial \theta_1},\frac{\partial\pi_{\theta}(a_2,s)}{\partial \theta_2},...,\frac{\partial\pi_{\theta}(a_2,s)}{\partial \theta_k}]+...+ t_M[\frac{\partial\pi_{\theta}(a_M,s)}{\partial \theta_1},\frac{\partial\pi_{\theta}(a_M,s)}{\partial \theta_2},...,\frac{\partial\pi_{\theta}(a_M,s)}{\partial \theta_k}] $

这个向量的维度为[1,k],再将求导结果乘上v

$ \frac{\partial J_t}{\partial \theta}v= t_1[\frac{\partial\pi_{\theta}(a_1,s)}{\partial \theta_1}v_1,\frac{\partial\pi_{\theta}(a_1,s)}{\partial \theta_2}v_2,...,\frac{\partial\pi_{\theta}(a_1,s)}{\partial \theta_k}v_k]+ t_2[\frac{\partial\pi_{\theta}(a_2,s)}{\partial \theta_1}v_1,\frac{\partial\pi_{\theta}(a_2,s)}{\partial \theta_2}v_2,...,\frac{\partial\pi_{\theta}(a_2,s)}{\partial \theta_k}v_k]+...+ t_M[\frac{\partial\pi_{\theta}(a_M,s)}{\partial \theta_1}v_1,\frac{\partial\pi_{\theta}(a_M,s)}{\partial \theta_2}v_2,...,\frac{\partial\pi_{\theta}(a_M,s)}{\partial \theta_k}v_k] $

再对t求导 得到的结果就是我们要求的东西[M,1]

$ \frac{\partial\frac{\partial J_t}{\partial \theta}}{\partial t}v =\begin{bmatrix} \frac{\partial\pi_{\theta}(a_1,s)}{\partial \theta_1}v_1+\frac{\partial\pi_{\theta}(a_1,s)}{\partial \theta_2}v_2+...+\frac{\partial\pi_{\theta}(a_1,s)}{\partial \theta_k}v_k\\ \frac{\partial\pi_{\theta}(a_2,s)}{\partial \theta_1}v_1 + \frac{\partial\pi_{\theta}(a_2,s)}{\partial \theta_2}v_2+... + \frac{\partial\pi_{\theta}(a_2,s)}{\partial \theta_k}v_k\\ \vdots \\ \frac{\partial\pi_{\theta}(a_M,s)}{\partial \theta_1}v_1 + \frac{\partial\pi_{\theta}(a_M,s)}{\partial \theta_2}v_2+...+ \frac{\partial\pi_{\theta}(a_M,s)}{\partial \theta_k}v_k \\ \end{bmatrix} $

接下来求

$ \begin{bmatrix} \frac{1}{\pi_{\theta_{old}}(a_1,s)}& 0... & 0\\ \vdots& \ddots & \\ 0& & \frac{1}{\pi_{\theta_{old}}(a_M,s)}\\ \end{bmatrix} \cdot \begin{bmatrix} \frac{\partial\pi_{\theta}(a_1,s)}{\partial \theta_1}& ... & \frac{\partial\pi_{\theta}(a_1,s)}{\partial \theta_k}\\ \vdots& \ddots & \\ \frac{\partial\pi_{\theta}(a_M,s)}{\partial \theta_1}& & \frac{\partial\pi_{\theta}(a_M,s)}{\partial \theta_k}\\ \end{bmatrix}\cdot \begin{bmatrix} v_1\\ \vdots\\ v_k\\ \end{bmatrix} $

已知

$ \frac{\partial\frac{\partial J_t}{\partial \theta}}{\partial t}v =\begin{bmatrix} \frac{\partial\pi_{\theta}(a_1,s)}{\partial \theta_1}v_1+\frac{\partial\pi_{\theta}(a_1,s)}{\partial \theta_2}v_2+...+\frac{\partial\pi_{\theta}(a_1,s)}{\partial \theta_k}v_k\\ \frac{\partial\pi_{\theta}(a_2,s)}{\partial \theta_1}v_1 + \frac{\partial\pi_{\theta}(a_2,s)}{\partial \theta_2}v_2+... + \frac{\partial\pi_{\theta}(a_2,s)}{\partial \theta_k}v_k\\ \vdots \\ \frac{\partial\pi_{\theta}(a_M,s)}{\partial \theta_1}v_1 + \frac{\partial\pi_{\theta}(a_M,s)}{\partial \theta_2}v_2+...+ \frac{\partial\pi_{\theta}(a_M,s)}{\partial \theta_k}v_k \\ \end{bmatrix} = \begin{bmatrix} \frac{\partial\pi_{\theta}(a_1,s)}{\partial \theta_1}& ... & \frac{\partial\pi_{\theta}(a_1,s)}{\partial \theta_k}\\ \vdots& \ddots & \\ \frac{\partial\pi_{\theta}(a_M,s)}{\partial \theta_1}& & \frac{\partial\pi_{\theta}(a_M,s)}{\partial \theta_k}\\ \end{bmatrix}\cdot \begin{bmatrix} v_1\\ \vdots\\ v_k\\ \end{bmatrix} $要计算的就是$ \begin{bmatrix} \frac{1}{\pi_{\theta_{old}}(a_1,s)}& 0... & 0\\ \vdots& \ddots & \\ 0& & \frac{1}{\pi_{\theta_{old}}(a_M,s)}\\ \end{bmatrix} \cdot \frac{\partial\frac{\partial J_t}{\partial \theta}}{\partial t}v $

因为是对角阵 dj..为列向量 可以将其写为对应元素相乘(element wise)的形式

$ [\frac{1}{\pi_{\theta_{old}}(a_1,s)},\frac{1}{\pi_{\theta_{old}}(a_2,s)},...,\frac{1}{\pi_{\theta_{old}}(a_M,s)}]\circ \frac{\partial\frac{\partial J_t}{\partial \theta}}{\partial t}v $,$ \circ $是element wise操作

$ =\begin{bmatrix} \frac{1}{\pi_{\theta_{old}}(a_1,s)}( \frac{\partial\pi_{\theta}(a_1,s)}{\partial \theta_1}v_1+\frac{\partial\pi_{\theta}(a_1,s)}{\partial \theta_2}v_2+...+\frac{\partial\pi_{\theta}(a_1,s)}{\partial \theta_k}v_k) \\ \frac{1}{\pi_{\theta_{old}}(a_2,s)}( \frac{\partial\pi_{\theta}(a_2,s)}{\partial \theta_1}v_1 + \frac{\partial\pi_{\theta}(a_2,s)}{\partial \theta_2}v_2+... + \frac{\partial\pi_{\theta}(a_2,s)}{\partial \theta_k}v_k)\\ \vdots \\ \frac{1}{\pi_{\theta_{old}}(a_M,s)}(\frac{\partial\pi_{\theta}(a_M,s)}{\partial \theta_1}v_1 + \frac{\partial\pi_{\theta}(a_M,s)}{\partial \theta_2}v_2+...+ \frac{\partial\pi_{\theta}(a_M,s)}{\partial \theta_k}v_k) \\ \end{bmatrix} $

设$ M=[\frac{1}{\pi_{\theta_{old}}(a_1,s)},\frac{1}{\pi_{\theta_{old}}(a_2,s)},...,\frac{1}{\pi_{\theta_{old}}(a_M,s)}] $，$ Jtv=\frac{\partial\frac{\partial J_t}{\partial \theta}}{\partial t}v $,$ MJtv=[\frac{1}{\pi_{\theta_{old}}(a_1,s)},\frac{1}{\pi_{\theta_{old}}(a_2,s)},...,\frac{1}{\pi_{\theta_{old}}(a_M,s)}]\circ \frac{\partial\frac{\partial J_t}{\partial \theta}}{\partial t}v $

最后

$ \begin{bmatrix} \frac{\partial\pi_{\theta}(a_1,s)}{\partial \theta_1}& ... & \frac{\partial\pi_{\theta}(a_M,s)}{\partial \theta_1}\\ \vdots& \ddots & \\ \frac{\partial\pi_{\theta}(a_1,s)}{\partial \theta_k}& & \frac{\partial\pi_{\theta}(a_M,s)}{\partial \theta_k}\\ \end{bmatrix}\cdot MJtv $

记$ MJtv_i=\frac{1}{\pi_{\theta_{old}}(a_i,s)}(\frac{\partial\pi_{\theta}(a_i,s)}{\partial \theta_1}v_1 + \frac{\partial\pi_{\theta}(a_i,s)}{\partial \theta_2}v_2+...+ \frac{\partial\pi_{\theta}(a_i,s)}{\partial \theta_k}v_k) $

$ =\begin{bmatrix} \frac{\partial\pi_{\theta}(a_1,s)}{\partial \theta_1}MJtv_1+ \frac{\partial\pi_{\theta}(a_2,s)}{\partial \theta_1}MJtv_2+...+ \frac{\partial\pi_{\theta}(a_M,s)}{\partial \theta_1}MJtv_M\\ \frac{\partial\pi_{\theta}(a_1,s)}{\partial \theta_2}MJtv_1+ \frac{\partial\pi_{\theta}(a_2,s)}{\partial \theta_2}MJtv_2+...+ \frac{\partial\pi_{\theta}(a_M,s)}{\partial \theta_2}MJtv_M \\ \vdots \\ \frac{\partial\pi_{\theta}(a_1,s)}{\partial \theta_k}MJtv_1+ \frac{\partial\pi_{\theta}(a_2,s)}{\partial \theta_k}MJtv_2+...+ \frac{\partial\pi_{\theta}(a_M,s)}{\partial \theta_k}MJtv_M\\ \end{bmatrix} $

对于这个结果的计算 我们一般先将MJtv先求出来

再计算 $ \sum_i\pi_{\theta}(a_i,s)\cdot MJtv_i $再将这个求和式对网络参数θ=[θ1,θ2,..,θk]

求偏导就得到了最终结果(注意在求导时要将MJtv脱离计算图因为MJtv中包含参数θ 但是我们不想MJtv对θ求导而只是想将它当为标量)

   def Fvp_fim(v):#快速求 Hv  海森阵和某个向量的乘积
        # M是[1/pi_old(a1),...,1/pi_old(aM)]
        # mu是[pi(a1),...,pi(aM)]
        M, mu, info = policy_net.get_fim(states)
        mu = mu.view(-1)
        filter_input_ids = set() if policy_net.is_disc_action else set([info['std_id']])

        t = ones(mu.size(), requires_grad=True, device=mu.device)# 全1
        # mu_t是sum i=1,...,M pi(ai)*ti
        mu_t = (mu * t).sum()
        # Jt 是sum i=1,...,M dpi(ai)*ti/dθ
        Jt = compute_flat_grad(mu_t, policy_net.parameters(), filter_input_ids=filter_input_ids, create_graph=True)
        # Jtv 是sum i=1,...,M dpi(ai)*ti/dθ *v
        Jtv = (Jt * v).sum()
        # Jt 是sum i=1,...,M dpi(ai)*ti/dθ *v /dt
        Jv = torch.autograd.grad(Jtv, t)[0]
        MJv = M * Jv.detach() # element wise
        mu_MJv = (MJv * mu).sum()
        JTMJv = compute_flat_grad(mu_MJv, policy_net.parameters(), filter_input_ids=filter_input_ids).detach()
        JTMJv /= states.shape[0] # 除 batch size
        if not policy_net.is_disc_action:
            std_index = info['std_index']
            JTMJv[std_index: std_index + M.shape[0]] += 2 * v[std_index: std_index + M.shape[0]]
        return JTMJv + v * damping
