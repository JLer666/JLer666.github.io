void Sampler::applyStartAndGoal(const State &start, const State &goal) {
  min_cost_ = goal.distanceFrom(start); // 终点和起点start之间的距离
  rotate_mat_ = calcRotationToWorldFlame(start, goal);

  auto center_v = ((start + goal) / 2).vals;
  center_v.push_back(0.0);
  center_state_ = Eigen::Map<Eigen::VectorXd>(&center_v[0], center_v.size());
}

State Sampler::run(const Sampler::Mode &mode) {
  State random_state(dim_);
  switch (mode) {
    case Mode::WholeArea: { // 在全局空间采样
    // 对每个维度进行均匀分布采样 作为此维度的状态采样值
      for (size_t i = 0; i < dim_; i++) {
        // dist_space_是一维均匀分布的数组   对dist_space_[i]--全局空间中的某个维度 进行均匀分布采样 作为此全局状态 此维度状态的值
        random_state.vals[i] = dist_space_[i](rand_);
      }
      break;
    }
    case Mode::HeuristicDomain: { // 在好奇区域内采样
      // definition of diagonal element
      // best_cost_ 椭圆长轴长,2a    min_cost_起点到终点距离--两个焦点长,2c
      auto diag_val = std::sqrt(std::pow(best_cost_, 2) - std::pow(min_cost_, 2)) / 2.0;// 短轴长b
      auto diag_v = std::vector<double>(dim_ + 1, diag_val);
      diag_v[0] = best_cost_ / 2.0;
      // 将所有长短轴写成一个对角阵L代表这个超椭球 L = diag{cmax/2,sqrt(cmax^2-cmin^2)/2,sqrt(cmax^2-cmin^2)/2
      // ,sqrt(cmax^2-cmin^2)/2,...} 状态维度数+1的对角阵  +1是因为增广

      // random sampling on unit n-ball
      auto x_ball_v = sampleUnitNBall(dim_).vals;// 在单位超球中随机采样一个状态维度的 随机状态
      x_ball_v.push_back(0.0);// 状态维度数+1  +1是因为增广

      // trans sampling pt
      // rotate_mat_*L(diag_v)*xBall(x_ball_v) + xCenter(center_state_)
      // L(diag_v)*xBall(x_ball_v)将单位超球上采样的点映射到超椭圆中
      // rotate_mat_*... + xCenter  将超椭圆中坐标映射到世界坐标中
      auto rand = rotate_mat_ * Eigen::Map<Eigen::VectorXd>(&*diag_v.begin(), diag_v.size()).asDiagonal() *
                      Eigen::Map<Eigen::VectorXd>(&*x_ball_v.begin(), x_ball_v.size()) +
                  center_state_;

      auto row_i = 0;
      for (auto &val : random_state.vals) {
        val = rand(row_i, 0);// rand在数据结构上是Eigen矩阵，但实际数学上算出来是个列向量 所以通过矩阵坐标索引
        //将列向量中的每行数据拿出来
        row_i++;
      }
      // 返回的是在超椭圆空间中的状态 并将状态转到世界坐标下返回
    } break;
  }

  return random_state;
}

std::vector<std::uniform_real_distribution<>> Sampler::generateSpaceDistribution(const EuclideanSpace &space) const {
  std::vector<std::uniform_real_distribution<double>> rand_restrictions;
  rand_restrictions.reserve(space.getDim());
  // space 保存了栅格空间的 各个维度上下限
  // 再用每个维度的上下限去初始化一个均匀分布
  // 之后在全局中的采样 就通过对每一个维度上的均匀分布来随机采样一个 此状态维度的随机值
  for (size_t di = 1; di <= space.getDim(); di++) {
    rand_restrictions.emplace_back(space.getBound(di).low, space.getBound(di).high);// emplace_back直接构造 一维均匀分布
    // 均匀分布参数[low,high]
  }
  return rand_restrictions;
}

  //                 [(self.goal.y - self.start.y) / cMin], [0]])

  // e_theta = math.atan2(a1[1], a1[0])
  // # first column of identity matrix transposed
  // id1_t = np.array([1.0, 0.0, 0.0]).reshape(1, 3)
  // M = a1 @ id1_t
  // U, S, Vh = np.linalg.svd(M, True, True)
  // C = np.dot(np.dot(U, np.diag(
  //     [1.0, 1.0, np.linalg.det(U) * np.linalg.det(np.transpose(Vh))])),
  //             Vh)
Eigen::MatrixXd Sampler::calcRotationToWorldFlame(const State &start, const State &goal) const {// 计算从椭圆坐标系到世界坐标系的旋转矩阵
  // 输入检查
  if (start.getDim() != goal.getDim() || start.getDim() < 2) {
    throw std::invalid_argument("[" + std::string(__PRETTY_FUNCTION__) + "] " + "State dimension is invalid");
  } else if (goal.distanceFrom(start) == 0) {
    throw std::invalid_argument("[" + std::string(__PRETTY_FUNCTION__) + "] " + "Start and goal are same state");
  }

  // 1 P(在世界坐标系下的表达)将此点集的归一化到0处 得到a1_state
  // 将向量的几何中心平移到原点 并且将点集(说是点集 其实只有一个点)范围归一化
  auto a1_state = (goal - start) / goal.distanceFrom(start);
  auto a1_v = a1_state.vals;// vals是state大小的vecyor
  a1_v.push_back(0.0);// 在state后面再加一个维度数据为0 -- 为了表达增广矩阵

//2 C = PQ^T
// 计算协方差矩阵
//  Eigen::MatrixXd::Identity(1, a1_v.size()) -- 单位矩阵的第一列T(注意只有一列)

// 两个点集的原点都移到xstart作为点集的原点!!!(消除平移的影响 因为只是计算旋转矩阵) 再都归一化
// P为以xstart为原点归一化世界坐标系下 xgoal状态坐标 Q 以xstart为原点归一化椭圆坐标系下 xgoal状态坐标
// P=(xgoal-xstart)/dist(goal2start) 世界坐标系横轴归一状态 Q = 单位矩阵的第一列T(注意只有一列)
  auto M = Eigen::Map<Eigen::VectorXd>(&*a1_v.begin(), a1_v.size()) * Eigen::MatrixXd::Identity(1, a1_v.size());//列向量asize行1列  1行 asize列 的行向量
  // 特征值和特征向量A nxn阵 Ax = /\x  /\为A的一个特征值 x为此特征值对应的特征向量
  // 如果我们已经求出了A的所有n个特征值，特征向量 p1,p2,...p3
  // 将p1..pn组成一个矩阵P  则A可以用特征值分解为 A = P S P^-1  S为由n个特征值组成的主对角线所形成的对角阵
  // 此时的P满足 PTP = I(单位阵)  即 PT = P^-1 P叫做酉矩阵
  // 所以A的特征值分解也可以写为A = P S PT
  // 注意A是方阵才能进行特征值分解，所以对于A非方阵的情况有SVD分解

  // A非方阵A mxn = U S VT   。U，V都是酉矩阵(即满足PT = P^-1) S是一个对角阵(只有主对角线上有值)
  // S主对角线上的每个值都叫做奇异值！！
  // ATA -- nxn 和AAT-- mxm一定是方阵 所以可以进行特征值分解
  // (ATA)vi = /\v vi  得到n个特征向量v
  // (AAT)ui = /\u ui  得到m个特征向量u
  //  A = U S VT -- AV = US 由于S是对角阵 可以得到  Avi = 奇异值i ui --   求解此方程得奇异值
  // 推论 特征值矩阵等于奇异值矩阵的平方  --- 奇异值i = sqrt(特征值)  这个特征值指的是ATA AAT方阵的特征值

  // 对于奇异值,它跟我们特征分解中的特征值类似，在奇异值矩阵主对角线中也是按照从大到小排列，
  // 而且奇异值的减少特别的快，
  // 在很多情况下，前10%甚至1%的奇异值的和就占了全部的奇异值之和的99%以上的比例。
  // 也就是说，我们也可以用最大的k个的奇异值和对应的左右奇异向量来近似描述矩阵。
  // A mxn = Umxm Smxn VT nxn 约等于 U mxk Skxk VT kxn
  //  也就是说一个很大的矩阵A可以用3个小矩阵UVS来保存节省了大量空间
  //3 V,S,W = SVD(C)
  //
  auto svd = Eigen::JacobiSVD<Eigen::MatrixXd>(M, Eigen::ComputeFullU | Eigen::ComputeFullV);// 将M阵SVD分解
  // 4 d = sign(det(C))
  auto diag_v = std::vector<double>(a1_v.size(), 1.0);
  // determinant是求行列式
  diag_v[diag_v.size() - 1] = svd.matrixV().determinant();
  diag_v[diag_v.size() - 2] = svd.matrixU().determinant();

  // 感觉他这里写错了！！！
  // 原版论文里是  C = U diag{1,1,..,det(U)*det(V)}V^T
  // diag_v[diag_v.size() - 1] = svd.matrixU().determinant()*svd.matrixV().determinant();

  // R = W [对角1...对角线上的最后一个元素为d] V^T
  return svd.matrixU() * Eigen::Map<Eigen::VectorXd>(&*diag_v.begin(), diag_v.size()).asDiagonal() *
         svd.matrixV().transpose();
}

// 在单位超球中采样一点
// http://extremelearning.com.au/how-to-generate-uniformly-random-points-on-n-spheres-and-n-balls/
// 类似其中的Method 7. Muller / Marsaglia (‘Normalized Gaussians’) 采样方案
// https://blogs.sas.com/content/iml/2016/04/06/generate-points-uniformly-in-ball.html
// 随机数采样方法 https://www.sciencedirect.com/science/article/pii/S0047259X10001211
State Sampler::sampleUnitNBall(const uint32_t &dim) {
  // 在单位超球内均匀地采样
  if (dim == 0) {
    throw std::invalid_argument("[" + std::string(__PRETTY_FUNCTION__) + "] " + "Can not sample zero-dimension ball");
  }
  // dist_gauss_(std::normal_distribution<>(0.0, 1.0)), // 一维正态分布
  // dist_unit_(std::uniform_real_distribution<>(0.0, 1.0)), // 一维均匀分布
  State x(dim);
  while (true) {
    for (auto &v : x.vals) {
      v = dist_gauss_(rand_); // rand_是随机种子 以随机种子在一维正态分布中采样一点值 0-1
    }
    // 采样了state维度个随机数 作为这个state向量的随机数x

    auto r = x.norm();// sqrt(x[0]^2 + x[1]^2 + ....) 原点到随机状态采样点 向量的长度
    if (r != 0.0) {// 浮点数不存在==
      x = x / r;// normalize 归一化向量x 使 原点到随机状态采样点 向量的长度为1 --- 最后的x一定是单位超球中的点
      break;
    }
  }
  // x是单位超球中的采样点

  // 在单位超球中创造一个均匀分布
  // std::pow(dist_unit_(rand_), 1.0 / dim)--均匀分布sample得到半径^dim次方
  // dist_unit_(rand_) -- 0-1
  // 采样出来的点依旧是在单位超球内的点 这个均匀分布相当于对半径进行放缩(随机均匀采样)从而能采样到0~r内的所有点
  // https://blogs.sas.com/content/iml/2016/04/06/generate-points-uniformly-in-ball.html
  // r (==1)* uni(0,1)^1/d*x
  return x * std::pow(dist_unit_(rand_), 1.0 / dim); // rand^1/dim
}
