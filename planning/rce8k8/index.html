struct KdTreeNode{

    void KdTree::SearchKdTreeNode(const geometry_msgs::Point& target,
                                KdTreeNode* current_node, // 当前递归深度的节点
                                KdTreeNode*& current_nearest_node,// 指针的引用 此指针可修改 并做传出参数
                                double& current_cloest_distance2,// 当前的最近欧式距离^2
                                int depth) const{
            // 搜索target节点的 最近邻居节点
            //                                                                     |
            //                                        右子树根节点 是半个子空间的中心   x划分
            //                                                                     |
            // ----------父节点 y划分是左右子空间的中心-----------
            //                                      |                                     target
            //    左子树根节点 是半个子空间的中心 x划分
            //                                      |

            // 可能出现target在 左或右某个子空间中  但是 它离另外一个子空间更近 所以需要检查是否离另一子空间中新更近
            // 如果没有离另一子空间更近 则那个子空间中不存在离target更近的节点?
            // 不太靠谱，还是有可能存在的，如果另一子空间中方差较大，就是可能存在 dis(子空间根节点,target) > dis(某一子节点,target)
            // 那根据dis(子空间根节点,target) > current_cloest_distance2 直接不看这个子空间 就可能错过最近节点
            // 之前没理解到位 对于这个是否要检查另一分支的条件没有搞清楚

            // 以target为圆心 以当前最近距离current_cloest_distance2为半径画圆
            // 如果圆与节点所代表的轴！！！相交  (则轴线两边的空间都需要检查 因为一边在向下递归中检查过了 所以)
            // 则需要将轴的另外一边需要检查
            // 判断相交 如果圆心target和轴线的垂直距离 小于最近距离 则就是发生相交
            // 在笔记中有图

            // 先一路到最底叶子节点 (根据那层的比较维度 二分地选择分支递归向下) 中间不断计算到节点距离 更新最近节点指针
            // 在向下的同时记录 以current_node节点为圆心 target到current_node节点的距离为半径，
            // 如果current_node的子节点 和 target之间的距离大于这个半径 则那个分支可以完全不看
            // 并且在之后回溯检查的时候也可以完全不管那些分支

            if(!current_node) return;//到底开始回溯

            //向下深入   查询与target处于同一子空间的节点   很快能找到最近邻的近似！！！点
            bool turn_left = (depth%2)? target.y <= current_node->point.y: target.x <= current_node->point.x;

            if(turn_left)
                SearchKdTreeNode(target,current_node->left_node,current_nearest_node,current_cloest_distance2,depth+1);
            else
                SearchKdTreeNode(target,current_node->right_node,current_nearest_node,current_cloest_distance2,depth+1);

            //回溯检查 另一分支 用turn_left就可直到上次哪个分支没被检查

            // 为什么不在向下深入的时候判断最近距离 而在回溯的时候计算最近距离

            // 并且因为下面 是否要查看另一分支时需要用到current_cloest_distance2，
            // 在回溯的时候检查 越接近根节点current_cloest_distance2越接近正确值的那个最小距离
            // 回溯到根节点时current_cloest_distance2越小，越有机会不去查看分支 可以减去大量的分支
            double r = (target.x - current_node->point.x) * (target.x - current_node->point.x) + (target.y - current_node->point.y) * (target.y- current_node->point.y);
            if( r < current_cloest_distance2)
            {
                current_cloest_distance2 = r;
                current_nearest_node = current_node;
            }
            // 检查另一分支 的距离是否比当前最近距离current_cloest_distance2还近 如果更近才有检查的价值

            // 理解错误  之前没理解到位 对于这个是否要检查另一分支的条件没有搞清楚
            // https://blog.csdn.net/silangquan/article/details/41483689  这个帖子 或者笔记有解释
            // if(turn_left)
            //     if(current_node->right_node)
            //         out_r = (target.x - current_node->right_node->point.x) * (target.x - current_node->right_node->point.x) +
            //         (target.y - current_node->right_node->point.y) * (target.y- current_node->right_node->point.y) > current_cloest_distance2? true:false;
            // else
            //     if(current_node->left_node)
            //         out_r = (target.x - current_node->left_node->point.x) * (target.x - current_node->left_node->point.x) +
            //         (target.y - current_node->left_node->point.y) * (target.y- current_node->left_node->point.y) > current_cloest_distance2? true:false;

            // if(!out_r) // 另一分支子空间超出范围则没有检查的必要 (超球面和子空间矩形完全不相交)
            // {
            //     if(turn_left)// 之前检查左 则这次检查右
            //         SearchKdTreeNode(target,current_node->right_node,current_nearest_node,current_cloest_distance2,depth+1);
            //     else // 之前检查右 则这次检查左
            //         SearchKdTreeNode(target,current_node->left_node,current_nearest_node,current_cloest_distance2,depth+1);
            // }

            // 什么情况下，kd tree在回溯时需要检查另一空间
            // 如上图，以target为圆心，以当前最近距离为半径，如果这个圆与中位数轴线相交，则这走线的两边空间内的节点都需要检查，
            // 因为在递归向下时已经检查过了一个分支，所以回溯时只需要检查另外一个分支即可,判断是否相交 -- 圆心到轴线距离 < 最近邻距离(半径) 那就是相交了。
            // 为什么和轴线相交了就需要查看轴线两边的空间点?
            // 因为只要与轴线相交了，轴线两边的空间就一定都会有一部分在圆内！ 在圆内的空间中如果有节点就会出现比当前最近邻还近的节点。
            double tar2axis_dis2 = !(depth%2)
                    ?   (target.x - current_node->point.x) * (target.x - current_node->point.x)
                    :   (target.y - current_node->point.y) * (target.y - current_node->point.y); //圆心到current_node轴线的距离

            //如果圆(tar为圆心 最近邻距离current_cloest_distance2为半径)与轴线相交 则需要检查另一分支
            if(current_cloest_distance2 > tar2axis_dis2)//相交
            {
                //检查另外一分支
                if(!turn_left)
                    SearchKdTreeNode(target,current_node->left_node,current_nearest_node,current_cloest_distance2,depth+1);
                else
                    SearchKdTreeNode(target,current_node->right_node,current_nearest_node,current_cloest_distance2,depth+1);
            }

    }

    // 搜索target  radius半径内的节点 结果存在near_nodes中传出
    void KdTree::SearchKdTreeNode(const geometry_msgs::Point& target,
                              KdTreeNode* current_node,// 当前递归深度的节点
                              std::vector<KdTreeNode*>& near_nodes,
                              double radius,
                              int depth) const{
             // 搜索target  radius半径内的节点 结果存在near_nodes中传出
            if(!current_node) return;//到底开始回溯

            //向下深入   查询与target处于同一子空间的节点   很快能找到最近邻的近似！！！点
            bool turn_left = (depth%2)? target.y <= current_node->point.y: target.x <= current_node->point.x;

            if(turn_left)
                SearchKdTreeNode(target,current_node->left_node,near_nodes,radius,depth+1);
            else
                SearchKdTreeNode(target,current_node->right_node,near_nodes,radius,depth+1);

            // 这里的距离计算在 回溯时或递归深入时计算都可以
            double r = (target.x - current_node->point.x) * (target.x - current_node->point.x) + (target.y - current_node->point.y) * (target.y- current_node->point.y);
            if(r < radius) near_nodes.push_back(current_node);// 满足半径内要求

            double tar2axis_dis2 = !(depth%2)
                    ?   (target.x - current_node->point.x) * (target.x - current_node->point.x)
                    :   (target.y - current_node->point.y) * (target.y - current_node->point.y); //圆心到current_node轴线的距离

            // 在target圆心radius半径 的圆是否与轴线相交
            // 只要与轴线相交了，轴线两边的空间就一定都会有一部分在圆内！  则那部分圆内空间就可能包含节点
            // 只要有节点 这些节点都是满足与target距离小于radius 的满足要求的节点
            if(radius > tar2axis_dis2)//相交
            {
                //检查另外一分支
                if(!turn_left)
                    SearchKdTreeNode(target,current_node->left_node,near_nodes,radius,depth+1);
                else
                    SearchKdTreeNode(target,current_node->right_node,near_nodes,radius,depth+1);
            }

    }

    // BFS直接遍历整棵树 得到离target最近的节点
    KdTreeNode* KdTree::BruceNearestKdTreeNode(const geometry_msgs::Point& target){
        KdTreeNode* nearest_node = nullptr;
        double min_distance2 = DBL_MAX;
        std::queue<KdTreeNode*> q;
        q.push(root_);
        while(!q.empty()){
            double distance2 = (target.x - q.front()->point.x)*(target.x - q.front()->point.x) + (target.y - q.front()->point.y)*(target.y - q.front()->point.y);
            if(distance2<min_distance2){
                min_distance2 = distance2;
                nearest_node = q.front();
            }
            if(q.front()->left_node) q.push(q.front()->left_node);
            if(q.front()->right_node) q.push(q.front()->right_node);
            q.pop();
        }
        return nearest_node;
    }

    // BFS直接遍历整棵树 得到离target radius范围内的节点
    std::vector<KdTreeNode*> KdTree::BruceNearKdTreeNode(const geometry_msgs::Point& target,
                                                                double radius){
        std::vector<KdTreeNode*> bruce_near_nodes;
        std::queue<KdTreeNode*> q;
        q.push(root_);
        while(!q.empty()){
            double distance2 = (target.x - q.front()->point.x)*(target.x - q.front()->point.x) + (target.y - q.front()->point.y)*(target.y - q.front()->point.y);
            if(distance2 < radius){
                bruce_near_nodes.push_back(q.front());
            }
            if(q.front()->left_node) q.push(q.front()->left_node);
            if(q.front()->right_node) q.push(q.front()->right_node);
            q.pop();
        }
        return bruce_near_nodes;
}

下面补充一下kd tree在高维度的改进 BBF

一节介绍的Kd-tree在维度较小时（例如：K≤30），算法的查找效率很高，然而当Kd-tree用于对高维数据（例如：K≥100）进行索引和查找时，就面临着维数灾难（curse of dimension）问题，查找效率会随着维度的增加而迅速下降。通常，实际应用中，我们常常处理的数据都具有高维的特点，例如在图像检索和识别中，每张图像通常用一个几百维的向量来表示，每个特征点的局部特征用一个高维向量来表征（例如：128维的SIFT特征）。因此，为了能够让Kd-tree满足对高维数据的索引，Jeffrey S. Beis和David G. Lowe提出了一种改进算法——Kd-tree with BBF（Best Bin First），该算法能够实现近似K近邻的快速搜索，在保证一定查找精度的前提下使得查找速度较快。 — 查找距离最近的特征向量

原始Kd-tree是为什么在低维空间中有效而到了高维空间后查找效率就会下降？

为了能够找到查询点Q在数据集合中的最近邻点，有一个重要的操作步骤：回溯，该步骤是在未被访问过的且与Q的超球面相交的子树分支中查找可能存在的最近邻点。随着维度K的增大，与Q的超球面相交的超矩形（子树分支所在的区域）就会增加，这就意味着需要回溯判断的树分支就会更多，从而算法的查找效率便会下降很大。

一个很自然的思路是：既然kd-tree算法在高维空间中是由于过多的回溯次数导致算法查找效率下降的话，我们就可以限制查找时进行回溯的次数上限，从而避免查找效率下降。这样做有两个问题需要解决：1）最大回溯次数怎么确定？2）怎样保证在最大回溯次数内找到的最近邻比较接近真实最近邻，即查找准确度不能下降太大。

问题1）：最大回溯次数怎么确定？

最大回溯次数一般人为设定，通常根据在数据集上的实验结果进行调整。

问题2）：怎样保证在最大回溯次数内找到的最近邻比较接近真实最近邻，即查找准确度不能下降太大？

限制回溯次数后，如果我们还是按照原来的回溯方法挨个地进行访问的话，那很显然最后的查找结果的精度就很大程度上取决于数据的分布和回溯次数了。挨个访问的方法的问题在于认为每个待回溯的树分支中存在最近邻的概率是一样的，所以它对所有的待回溯树分支一视同仁。实际上，在这些待回溯树分支中，有些树分支存在最近邻的可能性比其他树分支要高，因为树分支离Q点之间的距离或相交程度是不一样的，离Q更近的树分支存在Q的最近邻的可能性更高。因此，我们需要区别对待每个待回溯的树分支，即采用某种优先级顺序来访问这些待回溯树分支，使得在有限的回溯次数中找到Q的最近邻的可能性很高

基于BBF的Kd-Tree近似最近邻查找算法

已知：

Q：查询数据； KT：已建好的Kd-Tree；

  1. 查找Q的当前最近邻点P

1）从KT的根结点开始，将Q与中间结点node(k,m)进行比较，根据比较结果选择某个树分支Branch（或称为Bin）；并将未被选择的另一个树分支（Unexplored Branch）所在的树中位置和它跟Q之间的距离一起保存到一个优先级队列中Queue；

2）按照步骤1）的过程，对树分支Branch进行如上比较和选择，直至访问到叶子结点，然后计算Q与叶子结点中保存的数据之间的距离，并记录下最小距离D以及对应的数据P。

注：

A、Q与中间结点node(k,m)的比较过程：如果Q(k) > m则选择右子树，否则选择左子树。

B、优先级队列：按照距离从小到大的顺序排列。

C、叶子结点：每个叶子结点中保存的数据的个数可能是一个或多个。

  1. 基于BBF的回溯

已知：最大回溯次数BTmax

1）如果当前回溯的次数小于BTmax，且Queue不为空，则进行如下操作：

从Queue中取出最小距离对应的Branch，然后按照1.1步骤访问该Branch直至达到叶子结点；计算Q与叶子结点中各个数据间距离，如果有比D更小的值，则将该值赋给D，该数据则被认为是Q的当前近似最近邻点；

2）重复1）步骤，直到回溯次数大于BTmax或Queue为空时，查找结束，此时得到的数据P和距离D就是Q的近似最近邻点和它们之间的距离。
